    28	def build_tree():
    29	    tree = CommandTree()
    30	    catalog.register_all(tree)

    34	def _split_alias(argv):
    35	    """Remove a leading ``@alias`` word; return ``(alias, argv)``."""
    36	    for index, raw in enumerate(argv):
    37	        if raw == "--":
    38	            break
    39	        if raw.startswith("--") and len(raw) > 2:
    40	            continue
    41	        if aliases.is_alias_name(raw):
    42	            return raw, argv[:index] + argv[index + 1:]
    43	        break
    44	    return None, argv
    45	
    46	
    47	class _Invocation:
    48	    def __init__(self, argv, env, stdout, stderr, tree, cwd):
    49	        self.argv = argv
    50	        self.env = env
    51	        self.stdout = stdout
    52	        self.stderr = stderr
    53	        self.tree = tree
    54	        self.cwd = cwd
    55	
    56	    def start(self, alias):
    57	        out = Output(self.stdout, self.stderr)
    58	        try:
    59	            items = rewrites.apply(tokens.lex(self.argv))
    60	            self.runtime, self.items = runtime_config.extract(items, self.env)
    61	        except CliError as exc:
    62	            out.error(exc.message)
    63	            return exc.exit_code
    64	        self.out = out = Output(self.stdout, self.stderr,
    65	                                quiet=self.runtime.quiet(), debug=self.runtime.debug())
    66	        for warning in self.runtime.warnings:
    67	            out.warning(warning)
    68	        try:
    69	            self.locations = config_files.Locations(self.env)
    70	            self.loaded = config_files.load(self.locations, self.cwd)
    71	            self.registry = aliases.AliasRegistry(
    72	                self.loaded.aliases, aliases.runtime_aliases(self.env))
    73	            for path in self.loaded.files:
    74	                out.debug(f"Loaded config file: {path}")
    75	            return self.execute(alias, ())
    76	        except CliError as exc:
    77	            out.error(exc.message)
    78	            return exc.exit_code
    79	
    80	    def execute(self, alias, stack):
    81	        if alias is None:
    82	            return self.invoke({})
    83	        target = self.registry.resolve(alias)
    84	        if not target.is_group():
    85	            return self.invoke(target.connection)
    86	        code = 0
    87	        for member in target.members:
    88	            if member == alias or member in stack:
    89	                continue
    90	            self.out.log(member)
    91	            try:
    92	                result = self.execute(member, stack + (alias,))
    93	            except CliError as exc:
    94	                self.out.error(exc.message)
    95	                result = exc.exit_code
    96	            code = result or code
    97	        return code
    98	
    99	    def invoke(self, connection):
   100	        config = runtime_config.layer(
   101	            self.runtime, self.loaded.values, connection,
   102	            locations=self.locations, loaded=self.loaded, aliases=self.registry)
   103	        if config.ssh():
   104	            return remote.run(config.ssh(), self.argv, connection,
   105	                              self.runtime.values, config, self.env, self.out,
   106	                              self.stdout, self.stderr)
   107	        return self.run_local(config)
   108	
   109	    def run_local(self, config):
   110	        out = self.out
   111	        if self.tree is None:
   112	            self.tree = build_tree()
   113	        tree = self.tree
   114	        items = self.items
   115	        words = [t.text for t in items if not t.is_option()] or ["help"]
   116	        options = [(t.name, t.value) for t in items if t.is_option()]
   117	        node = None
   118	        try:
   119	            node, remaining = dispatch.resolve(tree, words, config.disabled_commands())
   120	            out.debug(f"Resolved command '{node.path}' with arguments {remaining}")
   121	            if not node.is_runnable():
   122	                out.usage(help_text.composite_usage(node))
   123	                return 0
   124	            bound = binding.bind(node.path, node.synopsis, node.defaults(),
   125	                                 remaining, options)
   126	            result = node.run(CommandContext(node, bound, out, config, tree))
   127	            return int(result or 0)
   128	        except MissingPositional:
   129	            out.usage(help_text.usage_line(node))
   130	            return 1
   131	
   132	
   133	def run(argv, env, stdout, stderr, tree=None, cwd=None):
   134	    """Run one invocation and return its exit code."""
   135	    alias, argv = _split_alias(list(argv))
   136	    invocation = _Invocation(argv, env, stdout, stderr, tree,
   137	                             cwd if cwd is not None else os.getcwd())
   138	    return invocation.start(alias)

     3	Secret: how a list of records is rendered in each supported output format.
     4	"""

     6	import csv
     7	import io
     8	import json
     9	
    10	_FORMATS = ("table", "json", "csv", "yaml", "ids", "count")
    11	
    12	
    13	def supported_formats():
    14	    return list(_FORMATS)
    15	
    16	
    17	def _table(records, fields):
    18	    rows = [[str(r.get(f, "")) for f in fields] for r in records]
    19	    widths = [max([len(f)] + [len(row[i]) for row in rows])
    20	              for i, f in enumerate(fields)]
    21	    lines = ["  ".join(c.ljust(w) for c, w in zip(fields, widths)).rstrip()]
    22	    lines += ["  ".join(c.ljust(w) for c, w in zip(row, widths)).rstrip()
    23	              for row in rows]
    24	    return "\n".join(lines)
    25	
    26	
    27	def _csv(records, fields):
    28	    buffer = io.StringIO()
    29	    writer = csv.writer(buffer, lineterminator="\n")
    30	    writer.writerow(fields)
    31	    for record in records:
    32	        writer.writerow([record.get(f, "") for f in fields])
    33	    return buffer.getvalue().rstrip("\n")
    34	
    35	
    36	def _yaml(records, fields):
    37	    lines = ["---"]
    38	    for record in records:
    39	        for index, field in enumerate(fields):
    40	            lead = "- " if index == 0 else "  "
    41	            lines.append(f"{lead}{field}: {json.dumps(record.get(field, ''))}")
    42	    return "\n".join(lines)
    43	
    44	
    45	def render(records, fields, fmt):
    46	    """Render ``records`` (dicts) restricted to ``fields`` in format ``fmt``.
    47	
    48	    Raises ``ValueError`` for an unsupported format.
    49	    """
    50	    fields = list(fields)
    51	    if fmt == "table":
    52	        return _table(records, fields)
    53	    if fmt == "json":
    54	        return json.dumps([{f: r.get(f) for f in fields} for r in records])
    55	    if fmt == "csv":
    56	        return _csv(records, fields)
    57	    if fmt == "yaml":
    58	        return _yaml(records, fields)
    59	    if fmt == "ids":
    60	        return " ".join(str(r.get(fields[0], "")) for r in records)
    61	    if fmt == "count":
    62	        return str(len(records))
    63	    raise ValueError(fmt)
    64

    73	def render_mapping(mapping, fmt):
    74	    """Render a single key/value mapping. Raises ``ValueError`` for an
    75	    unsupported format."""
    76	    if fmt == "list":
    77	        width = max((len(k) for k in mapping), default=0)
    78	        return "\n".join(f"{k.ljust(width)}  {v}" for k, v in mapping.items())
    79	    if fmt == "json":
    80	        return json.dumps(mapping)
    81	    if fmt == "yaml":
    82	        return "\n".join(["---"] + [f"{k}: {json.dumps(v)}" for k, v in mapping.items()])
    83	    raise ValueError(fmt)

     1	"""Command resolution.
     2	
     3	Secret: how positional words are walked down the command tree, and when
     4	leftover words become arguments of the matched command rather than unknown
     5	subcommands.
     6	"""
     7	
     8	from . import messages as _messages
     9	from .errors import CliError
    10	from .suggest import suggest
    11	
    12	
    13	def resolve(tree, words, disabled=()):
    14	    """Find the deepest command matching the leading ``words``.
    15	
    16	    Returns ``(node, remaining_words)``. Raises ``CliError`` for an unknown
    17	    top-level command, an unknown subcommand under a group that cannot
    18	    itself run, or a command whose path is listed in ``disabled``.
    19	    """
    20	    disabled = set(disabled)
    21	    node = tree.root
    22	    words = list(words)
    23	    consumed = 0
    24	    for word in words:
    25	        if not node.has_children():
    26	            break
    27	        child = node.find_child(word)
    28	        if child is None:
    29	            if node.is_runnable() and not node.is_root():
    30	                break
    31	            suggestion = suggest(word, node.child_names())
    32	            if node.is_root():
    33	                raise CliError(_messages.unregistered_command(word, suggestion))
    34	            raise CliError(_messages.unregistered_subcommand(
    35	                word, node.path, suggestion))
    36	        node = child
    37	        consumed += 1
    38	        if node.path in disabled:
    39	            raise CliError(_messages.command_disabled(node.path))
    40	    return node, words[consumed:]

   111	def extract(items, env):
   112	    """Split global parameters out of ``items``.
   113	
   114	    Returns ``(RuntimeValues, remaining_tokens)``. In strict mode only tokens
   115	    before the first positional word are considered global.
   116	    """
   117	    strict = bool(env.get(_STRICT_ENV, ""))
   118	    values = {}
   119	    warnings = []
   120	    remaining = []
   121	    seen_positional = False
   122	    for token in items:
   123	        if not token.is_option():
   124	            seen_positional = True
   125	            remaining.append(token)
   126	            continue
   127	        if strict and seen_positional:
   128	            remaining.append(token)
   129	            continue
   130	        key, value, status = _GLOBALS.interpret(token.name, token.value)
   131	        if status == Synopsis.UNKNOWN:
   132	            remaining.append(token)
   133	        elif status == Synopsis.OK:
   134	            values[key] = value
   135	            if config_spec.is_deprecated(key):
   136	                warnings.append(_messages.deprecated_global_parameter(key))
   137	        else:
   138	            raise CliError(_messages.global_parameter_error(
   139	                key, needs_value=(status == Synopsis.NEEDS_VALUE)))
   140	    return RuntimeValues(values, strict, list(dict.fromkeys(warnings))), remaining

   143	def layer(runtime, file_values=None, alias_values=None, **context):
   144	    """Combine the sources: runtime > alias > config files > defaults."""
   145	    values = {}
   146	    for key, value in (file_values or {}).items():
   147	        param = config_spec.get(key)
   148	        if param is not None and param.file:
   149	            values[key] = value
   150	    for key, value in (alias_values or {}).items():
   151	        if config_spec.get(key) is not None and key not in ("debug", "quiet"):
   152	            values[key] = value
   153	    values.update(runtime.values)

     4	``--flag``) and positional words, including the ``--`` terminator.

     8	class Token:
     9	    """One lexical unit of the command line."""
    10	
    11	    __slots__ = ("_text", "_name", "_value", "_option")
    12	
    13	    def __init__(self, text, name=None, value=None, option=False):
    14	        self._text = text
    15	        self._name = name
    16	        self._value = value
    17	        self._option = option

    19	    def is_option(self):
    20	        return self._option

    30	        return self._name

    32	    @property

    34	        """Option value as a string, or ``None`` when no ``=`` was given."""
    35	        return self._value

    38	        return f"Token({self._text!r})"

    41	def positional(text):
    42	    """Build a positional word token."""
    43	    return Token(text)

    46	def lex(argv):
    47	    """Split raw argv strings into tokens."""
    48	    result = []
    49	    options_ended = False
    50	    for raw in argv:
    51	        if options_ended:
    52	            result.append(positional(raw))
    53	        elif raw == "--":
    54	            options_ended = True
    55	        elif raw.startswith("--") and len(raw) > 2:
    56	            body = raw[2:]
    57	            if "=" in body:
    58	                name, value = body.split("=", 1)
    59	            else:
    60	                name, value = body, None
    61	            result.append(Token(raw, name, value, option=True))
    62	        else:
    63	            result.append(positional(raw))
    64	    return result

    14	class BoundArguments:
    15	    """Validated arguments with defaults applied."""

    27	            return self._positionals[index]
    28	        return default

    31	        return self._options.get(name, default)

    40	def _check_positionals(synopsis, words):
    41	    if len(words) < synopsis.required_positional_count():
    42	        raise MissingPositional()
    43	    limit = synopsis.positional_limit()
    44	    if limit is not None and len(words) > limit:
    45	        raise CliError(_messages.too_many_positionals(words[limit:]))
    46	
    47	
    48	def _bind_options(synopsis, options):
    49	    bound = {}
    50	    needs_value, takes_no_value, unknown = [], [], []
    51	    for name, value in options:
    52	        key, interpreted, status = synopsis.interpret(name, value)
    53	        if status == Synopsis.OK:
    54	            bound[key] = interpreted
    55	        elif status == Synopsis.NEEDS_VALUE:
    56	            bound.pop(key, None)
    57	            needs_value.append(key)
    58	        elif status == Synopsis.TAKES_NO_VALUE:
    59	            takes_no_value.append(key)
    60	        elif name not in unknown:
    61	            unknown.append(name)
    62	
    63	    required = synopsis.required_option_names()
    64	    errors = [_messages.missing_parameter(key)
    65	              for key in required if key not in bound]
    66	    errors += [_messages.parameter_needs_value(key)
    67	               for key in dict.fromkeys(needs_value)
    68	               if key not in required and key not in bound]
    69	    errors += [_messages.parameter_takes_no_value(key)
    70	               for key in dict.fromkeys(takes_no_value)]
    71	    generic = synopsis.requires_generic_option()
    72	    if generic and not any(k not in synopsis.option_names() for k in bound):
    73	        errors.append(_messages.missing_parameter(generic))
    74	    known = synopsis.option_names()
    75	    errors += [_messages.unknown_parameter(key, suggest(key, known))
    76	               for key in unknown]
    77	    if errors:
    78	        raise CliError(_messages.parameter_errors(errors))
    79	    return bound
    80	
    81	
    82	def _apply_defaults(synopsis, words, options, defaults):
    83	    words = list(words)
    84	    names = synopsis.positional_names()
    85	    for name in names[len(words):]:
    86	        if name not in defaults:
    87	            break
    88	        words.append(defaults[name])
    89	    for name in synopsis.option_names():
    90	        if name not in options and name in defaults:
    91	            options[name] = defaults[name]
    92	    return words, options

    95	def bind(path, synopsis, defaults, words, options):
    96	    """Validate ``words`` and ``options`` against ``synopsis``.
    97	
    98	    ``options`` is a sequence of ``(name, value)`` pairs where ``value`` is a
    99	    string or ``None`` (no ``=`` given). Raises ``MissingPositional`` when
   100	    required positionals are absent and ``CliError`` for other problems.
   101	    """
   102	    malformed = synopsis.malformed_parts()
   103	    if malformed:
   104	        raise CliError(_messages.invalid_synopsis(path, malformed))
   105	    _check_positionals(synopsis, list(words))
   106	    bound = _bind_options(synopsis, options)
   107	    words, bound = _apply_defaults(synopsis, words, bound, dict(defaults))
   108	    return BoundArguments(words, bound)
