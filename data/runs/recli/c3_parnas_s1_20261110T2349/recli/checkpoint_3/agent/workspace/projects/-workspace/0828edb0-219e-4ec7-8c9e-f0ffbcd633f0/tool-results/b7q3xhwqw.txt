    14	class CommandNode:
    15	    """One command (or the root) in the command tree."""
    16	
    17	    def __init__(self, name, parent=None):

    19	        self._parent = parent
    20	        self._children = {}
    21	        self._aliases = {}
    22	        self._handler = None

    24	        self._synopsis = Synopsis("")
    25	        self._description = ""
    26	        self._alias = None
    27	        self._defaults = {}
    28	
    29	    def _configure(self, handler, synopsis, description, alias, defaults):
    30	        self._handler = handler
    31	        self._synopsis_text = synopsis or ""
    32	        self._synopsis = Synopsis(self._synopsis_text)
    33	        self._description = description or ""
    34	        self._alias = alias
    35	        self._defaults = dict(defaults or {})

    37	    @property
    38	    def name(self):
    39	        return self._name

    41	    @property
    42	    def path(self):
    43	        """Full space-separated command path ('' for the root)."""
    44	        words = []
    45	        node = self
    46	        while node._parent is not None:
    47	            words.append(node._name)
    48	            node = node._parent

    51	    @property
    52	    def parent(self):

    70	    def is_root(self):
    71	        return self._parent is None
    72	
    73	    def is_runnable(self):
    74	        return self._handler is not None
    75	
    76	    def has_children(self):
    77	        return bool(self._children)
    78	
    79	    def children(self):
    80	        return [self._children[n] for n in sorted(self._children, key=_sort_key)]
    81	
    82	    def child_names(self):
    83	        return sorted(self._children, key=_sort_key)

    85	    def find_child(self, word):
    86	        """Match ``word`` against child names, then child aliases."""
    87	        if word in self._children:
    88	            return self._children[word]
    89	        canonical = self._aliases.get(word)
    90	        return self._children.get(canonical) if canonical else None
    91	
    92	    def run(self, context):
    93	        return self._handler(context)
    94	
    95	    def _child(self, name, create):
    96	        if name not in self._children and create:
    97	            if name in self._aliases:
    98	                raise ValueError(f"'{name}' is already an alias under '{self.path}'")
    99	            self._children[name] = CommandNode(name, self)
   100	        return self._children.get(name)
   101	
   102	    def _add_alias(self, alias, canonical):
   103	        if alias in self._children or self._aliases.get(alias, canonical) != canonical:
   104	            raise ValueError(f"alias '{alias}' conflicts under '{self.path}'")
   105	        self._aliases[alias] = canonical

   108	class CommandTree:
   109	    """Registry of all commands."""
   110	
   111	    def __init__(self):
   112	        self._root = CommandNode("")
   113	
   114	    @property
   115	    def root(self):
   116	        return self._root
   117	
   118	    def register(self, path, handler=None, *, synopsis="", description="",
   119	                 alias=None, defaults=None):
   120	        """Register (or update) the command at space-separated ``path``.
   121	
   122	        Missing parent commands are created as non-runnable groups.
   123	        """
   124	        words = path.split()
   125	        if not words:
   126	            raise ValueError("command path must not be empty")
   127	        node = self._root
   128	        for word in words:
   129	            node = node._child(word, create=True)
   130	        node._configure(handler, synopsis, description, alias, defaults)
   131	        if alias:
   132	            node.parent._add_alias(alias, node.name)
   133	        return node
   134	
   135	    def find(self, words):
   136	        """Return the node at exactly ``words`` (aliases allowed) or ``None``."""
   137	        node = self._root
   138	        for word in words:
   139	            node = node.find_child(word)
   140	            if node is None:
   141	                return None
   142	        return None if node is self._root else node

     2	
     3	Secret: the textual grammar of synopsis strings and how each part is
     4	classified (positional, keyed parameter, flag, negatable flag, generic keyed

     8	import re
     9	
    10	_NAME = r"[A-Za-z0-9][\w-]*"
    11	_PLACEHOLDER = r"[\w|.-]+"
    12	
    13	_POSITIONAL = re.compile(rf"^<({_PLACEHOLDER})>$")
    14	_GENERIC = re.compile(rf"^--<({_PLACEHOLDER})>=<({_PLACEHOLDER})>$")
    15	_KEYED = re.compile(rf"^--({_NAME})=<({_PLACEHOLDER})>$")
    16	_KEYED_OPTIONAL_VALUE = re.compile(rf"^--({_NAME})\[=<({_PLACEHOLDER})>\]$")
    17	_FLAG = re.compile(rf"^--({_NAME})$")
    18	_NEGATABLE_FLAG = re.compile(rf"^--\[no-\]({_NAME})$")
    19	
    20	_POSITIONAL_KIND = "positional"
    21	_KEYED_KIND = "keyed"
    22	_FLAG_KIND = "flag"
    23	_GENERIC_KIND = "generic"
    24	_MALFORMED_KIND = "malformed"

    26	_REPEAT = "..."

    29	class _Part:
    30	    __slots__ = ("text", "kind", "name", "optional", "repeating", "value_optional")
    31	
    32	    def __init__(self, text, kind, name=None, optional=False, repeating=False,
    33	                 value_optional=False):
    34	        self.text = text
    35	        self.kind = kind
    36	        self.name = name
    37	        self.optional = optional
    38	        self.repeating = repeating
    39	        self.value_optional = value_optional

    42	def _outer_brackets(text):
    43	    """True when the leading '[' is closed by the trailing ']'."""
    44	    if not (text.startswith("[") and text.endswith("]")):
    45	        return False
    46	    depth = 0
    47	    for index, char in enumerate(text):
    48	        if char == "[":
    49	            depth += 1
    50	        elif char == "]":
    51	            depth -= 1
    52	            if depth == 0 and index != len(text) - 1:
    53	                return False
    54	    return depth == 0
    55	
    56	
    57	def _parse_part(text):
    58	    core = text
    59	    optional = repeating = False
    60	    if core.endswith(_REPEAT):
    61	        repeating, core = True, core[: -len(_REPEAT)]
    62	    if _outer_brackets(core):
    63	        optional, core = True, core[1:-1]
    64	        if core.endswith(_REPEAT):
    65	            repeating, core = True, core[: -len(_REPEAT)]
    66	
    67	    def part(kind, name=None, value_optional=False):
    68	        return _Part(text, kind, name, optional, repeating, value_optional)
    69	
    70	    match = _POSITIONAL.match(core)
    71	    if match:
    72	        return part(_POSITIONAL_KIND, match.group(1))
    73	    match = _GENERIC.match(core)
    74	    if match:
    75	        return part(_GENERIC_KIND, match.group(1))
    76	    match = _KEYED.match(core)
    77	    if match:
    78	        return part(_KEYED_KIND, match.group(1))
    79	    match = _KEYED_OPTIONAL_VALUE.match(core)
    80	    if match:
    81	        return part(_KEYED_KIND, match.group(1), value_optional=True)
    82	    match = _FLAG.match(core) or _NEGATABLE_FLAG.match(core)
    83	    if match:
    84	        return part(_FLAG_KIND, match.group(1))
    85	    return _Part(text, _MALFORMED_KIND)

    88	class Synopsis:
    89	    """A parsed synopsis that answers questions about accepted arguments."""
    90	
    91	    OK = "ok"
    92	    UNKNOWN = "unknown"
    93	    NEEDS_VALUE = "needs-value"
    94	    TAKES_NO_VALUE = "takes-no-value"
    95	
    96	    def __init__(self, text=""):
    97	        self._parts = [_parse_part(p) for p in (text or "").split()]
    98	        self._options = {}
    99	        for part in self._parts:
   100	            if part.kind in (_KEYED_KIND, _FLAG_KIND):
   101	                self._options[part.name] = part
   102	
   103	    def text(self):
   104	        return " ".join(part.text for part in self._parts)
   105	
   106	    def malformed_parts(self):
   107	        return [p.text for p in self._parts if p.kind == _MALFORMED_KIND]
   108	
   109	    def _positionals(self):
   110	        return [p for p in self._parts if p.kind == _POSITIONAL_KIND]
   111	
   112	    def required_positional_count(self):
   113	        return sum(1 for p in self._positionals() if not p.optional)
   114	
   115	    def positional_limit(self):
   116	        """Maximum number of positional words, or ``None`` if unbounded."""
   117	        positionals = self._positionals()
   118	        if any(p.repeating for p in positionals):
   119	            return None
   120	        return len(positionals)
   121	
   122	    def positional_names(self):
   123	        return [p.name for p in self._positionals()]
   124	
   125	    def option_names(self):
   126	        """Declared keyed-parameter and flag names, in declaration order."""
   127	        return list(self._options)
   128	
   129	    def required_option_names(self):
   130	        return [name for name, p in self._options.items() if not p.optional]
   131	
   132	    def accepts_any_option(self):
   133	        return any(p.kind == _GENERIC_KIND for p in self._parts)
   134	
   135	    def requires_generic_option(self):
   136	        """Name of a mandatory generic field placeholder, or ``None``."""
   137	        for p in self._parts:
   138	            if p.kind == _GENERIC_KIND and not p.optional:
   139	                return p.name
   140	        return None
   141	
   142	    def interpret(self, name, value):
   143	        """Interpret a supplied option.
   144	
   145	        ``value`` is the supplied string or ``None`` when no ``=`` was given.
   146	        Returns ``(key, value, status)`` where status is one of the class
   147	        constants. Flags yield booleans; keyed parameters yield strings.
   148	        """
   149	        part = self._options.get(name)
   150	        if part is not None:
   151	            if part.kind == _FLAG_KIND:
   152	                if value is None:
   153	                    return name, True, self.OK
   154	                return name, None, self.TAKES_NO_VALUE
   155	            if value is None and part.value_optional:
   156	                return name, True, self.OK
   157	            if value is None or value == "":
   158	                return name, None, self.NEEDS_VALUE
   159	            return name, value, self.OK
   160	        if name.startswith("no-") and value is None:
   161	            negated = self._options.get(name[3:])
   162	            if negated is not None and negated.kind == _FLAG_KIND:
   163	                return negated.name, False, self.OK
   164	        if self.accepts_any_option():
   165	            return name, (True if value is None else value), self.OK
   166	        return name, None, self.UNKNOWN

   103	class LoadedConfig:
   104	    """Values, aliases and other keys collected from config files."""
   105	
   106	    def __init__(self):
   107	        self.values = {}

   114	    def merge_file(self, path, tolerant=False, chain=()):
   115	        real = os.path.realpath(path)
   116	        if real in chain:
   117	            return
   118	        data = _read(path, tolerant)
   119	        if data is None:
   120	            return
   121	        meta = data.get(_META_KEY)
   122	        meta = meta if isinstance(meta, dict) else {}
   123	        base = os.path.dirname(os.path.abspath(path))
   124	        inherit = meta.get("inherit")
   125	        if isinstance(inherit, str) and inherit:
   126	            self.merge_file(_inherit_path(inherit, base), tolerant, chain + (real,))
   127	        merge = bool(meta.get("merge"))
   128	        self.files.append(path)
   129	        for key, value in data.items():
   130	            if key == _META_KEY:
   131	                continue
   132	            key = str(key)
   133	            if is_alias_name(key):
   134	                self._merge_alias(key, value, base, merge)
   135	            elif config_spec.file_param(key) is not None:
   136	                self._merge_param(key, value, base, merge)
   137	            else:
   138	                self.extra[key] = _combine(self.extra.get(key), value, merge)

   141	        if isinstance(value, list):
   142	            definition = [str(v) for v in value if v is not None]

   174	    """Write missing auto-created keys into the global config file."""
   175	    path = locations.global_config_path
   176	    try:
   177	        text = ""

   205	            raise CliError(messages.config_unwritable(path, exc)) from None

    46	def remote_command(args, bits, env):
    47	    """The shell command run on the remote side."""
    48	    parts = []
    49	    pre = (env.get(_PRE_CMD_ENV) or "").strip().rstrip(";").strip()
    50	    if pre:

    69	def _strip_comment(line):

    72	        if quote:
    73	            if char == quote:
    74	                quote = None
    75	            elif char == "\\" and quote == '"':

    84	def _read_quoted(text, start):
    85	    """Parse a quoted string starting at ``start``; return (value, end)."""
    86	    quote = text[start]
    87	    index = start + 1
    88	    chars = []
    89	    while index < len(text):
    90	        char = text[index]
    91	        if quote == "'" and char == "'":
    92	            if text[index + 1:index + 2] == "'":
    93	                chars.append("'")
    94	                index += 2
    95	                continue
    96	            return "".join(chars), index + 1
    97	        if quote == '"' and char == "\\":
    98	            code = text[index + 1:index + 2]
    99	            width = {"x": 2, "u": 4, "U": 8}.get(code)
   100	            if width:
   101	                try:
   102	                    chars.append(chr(int(text[index + 2:index + 2 + width], 16)))
   103	                except ValueError:
   104	                    raise YamlError("invalid escape sequence") from None
   105	                index += 2 + width
   106	                continue
   107	            chars.append(_ESCAPES.get(code, code))
   108	            index += 2
   109	            continue
   110	        if quote == '"' and char == '"':
   111	            return "".join(chars), index + 1
   112	        chars.append(char)
   113	        index += 1

   146	class _Flow:
   147	    def __init__(self, text):
   148	        self.text = text
   149	        self.pos = 0
   150	
   151	    def skip(self):
   152	        while self.pos < len(self.text) and self.text[self.pos] in " \t":
   153	            self.pos += 1
   154	
   155	    def peek(self):
   156	        self.skip()
   157	        return self.text[self.pos] if self.pos < len(self.text) else ""
   158	
   159	    def value(self, stops):
   160	        char = self.peek()
   161	        if char == "[":
   162	            return self.sequence()
   163	        if char == "{":
   164	            return self.mapping()
   165	        if char in "\"'":
   166	            value, self.pos = _read_quoted(self.text, self.pos)
   167	            return value
   168	        start = self.pos
   169	        while self.pos < len(self.text):
   170	            char = self.text[self.pos]
   171	            if char in stops:
   172	                break
   173	            if char == ":" and ":" in stops and (
   174	                    self.pos + 1 == len(self.text) or self.text[self.pos + 1] in " ,]}"):
   175	                break
   176	            self.pos += 1
   177	        return _resolve(self.text[start:self.pos])
   178	
   179	    def expect(self, char):
   180	        if self.peek() != char:
   181	            raise YamlError(f"expected '{char}' in flow collection")
   182	        self.pos += 1
   183	
   184	    def sequence(self):
   185	        self.expect("[")
   186	        items = []
   187	        while self.peek() != "]":
   188	            if not self.peek():
   189	                raise YamlError("unterminated flow sequence")
   190	            items.append(self.value(",]"))
   191	            if self.peek() == ",":
   192	                self.pos += 1
   193	        self.pos += 1
   194	        return items
   195	
   196	    def mapping(self):
   197	        self.expect("{")
   198	        result = {}
   199	        while self.peek() != "}":
   200	            if not self.peek():
   201	                raise YamlError("unterminated flow mapping")
   202	            key = self.value(":,}")
   203	            value = None
   204	            if self.peek() == ":":
   205	                self.pos += 1
   206	                value = self.value(",}")
   207	            result[key] = value
   208	            if self.peek() == ",":
   209	                self.pos += 1
   210	        self.pos += 1
   211	        return result

   215	    text = text.strip()

   228	class _Parser:
   229	    def __init__(self, text):
   230	        self.lines = []
   231	        for raw in text.splitlines():
   232	            if raw.strip() in ("---", "..."):
   233	                continue
   234	            content = _strip_comment(raw)
   235	            if not content.strip():
   236	                continue
   237	            if "\t" in content[:len(content) - len(content.lstrip())]:
   238	                raise YamlError("tabs are not allowed for indentation")
   239	            stripped = content.lstrip(" ")
   240	            self.lines.append([len(content) - len(stripped), stripped])
   241	
   242	    def parse(self):
   243	        if not self.lines:
   244	            return None
   245	        if len(self.lines) == 1 and _split_key(self.lines[0][1]) is None \
   246	                and not _is_seq_item(self.lines[0][1]):
   247	            return _inline(self.lines[0][1])
   248	        value, index = self.block(0, self.lines[0][0])
   249	        if index < len(self.lines):
   250	            raise YamlError(f"unexpected content: {self.lines[index][1]!r}")
   251	        return value
   252	
   253	    def block(self, index, indent):
   254	        if _is_seq_item(self.lines[index][1]):
   255	            return self.sequence(index, indent)
   256	        return self.mapping(index, indent)
   257	
   258	    def nested(self, index, indent, allow_same_indent_seq):
   259	        """Parse the block following a line ending in ':' or '-'."""
   260	        if index < len(self.lines):
   261	            next_indent, content = self.lines[index]
   262	            if next_indent > indent or (
   263	                    allow_same_indent_seq and next_indent == indent
   264	                    and _is_seq_item(content)):
   265	                return self.block(index, next_indent)
   266	        return None, index
   267	
   268	    def sequence(self, index, indent):
   269	        items = []
   270	        while index < len(self.lines):
   271	            line_indent, content = self.lines[index]
   272	            if line_indent != indent or not _is_seq_item(content):
   273	                break
   274	            rest = content[1:].lstrip(" ")

    16	        current = [i]

    28	    """Return the best candidate for a mistyped ``target``, or ``None``."""
    29	    pool = sorted(set(candidates), key=lambda c: (c.casefold(), c))

import io

import sys

from appctl_core import yaml_lite
from appctl_core.app import run

ROOT = os.path.dirname(os.path.dirname(os.path.abspath(__file__)))


@pytest.fixture
def home(tmp_path):
    root = tmp_path / "root"
    project = tmp_path / "project"
    project.mkdir()
    return root, project


def call(*argv, root, cwd, env=None):
    out, err = io.StringIO(), io.StringIO()
    full_env = {"APPCTL_ROOT_DIR": str(root), "PATH": os.environ.get("PATH", "")}
    full_env.update(env or {})
    code = run(list(argv), full_env, out, err, cwd=str(cwd))
    return code, out.getvalue(), err.getvalue()


def info(root, cwd, *extra, env=None):
    code, out, err = call("cli", "info", "--format=json", *extra,
                          root=root, cwd=cwd, env=env)
    assert code == 0, err
    return json.loads(out)


def write(path, text):
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(text)


# --- loading and precedence ------------------------------------------------

def test_global_config_auto_created(home):
    root, project = home
    call("cli", "version", root=root, cwd=project)
    data = yaml_lite.load((root / "config" / "config.yml").read_text())
    assert data == {"locale": "en_US", "installer_version": "stable"}


def test_auto_create_preserves_existing_values(home, tmp_path):
    root, project = home
    config = tmp_path / "custom.yml"
    write(config, "color: false\nlocale: fr_FR")
    call("cli", "version", root=root, cwd=project,
         env={"APPCTL_CONFIG_PATH": str(config)})
    assert yaml_lite.load(config.read_text()) == {
        "color": False, "locale": "fr_FR", "installer_version": "stable"}
