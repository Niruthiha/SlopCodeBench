"""Pipeline wiring.

Secret: the order in which the stages run (alias split, lex, rewrite,
extract globals, load config, expand aliases, resolve, bind, run or hand
off to a remote transport) and how failures map to output and exit codes.
"""

_COMPLETION = ["cli", "completion"]


class CommandContext:
    """Everything a command handler may use."""

    def __init__(self, command, args, out, config, tree, cwd=None, cache=None,
                 launcher=None):
        self.command = command
        self.args = args
        self.out = out
        self.config = config
        self.tree = tree
        self.cwd = cwd if cwd is not None else os.getcwd()
        self.cache = cache
        self.launcher = launcher

    def run_process(self, command, **kwargs):
        """Run an external command (see ``process.run``)."""
        kwargs.setdefault("out", self.out)
        kwargs.setdefault("cwd", self.cwd)
        return process.run(command, **kwargs)

    def launch_self(self, command, *args, **kwargs):
        """Re-dispatch an appctl command in a child process (see
        ``process.Launcher.launch_self``)."""
        return self.launcher.launch_self(command, *args, **kwargs)

    def runcommand(self, command, **kwargs):
        """Run an appctl command in a child process and return its output
        (see ``process.Launcher.runcommand``)."""
        return self.launcher.runcommand(command, **kwargs)

    def read_value(self, index):
        """Positional ``index``, or STDIN when it was not given, decoded
        according to ``--format``; ``None`` when neither is available."""
        raw = self.args.positional(index)
        if raw is None:
            raw = self.out.read_stdin()
            if raw is None:
                return None
        return values.read_value(raw, self.args.option("format"))


def build_tree():
    tree = CommandTree()
    catalog.register_all(tree)
    return tree


def _split_alias(argv):
    """Remove a leading ``@alias`` word; return ``(alias, argv)``."""
    for index, raw in enumerate(argv):
        if raw == "--":
            break
        if raw.startswith("--") and len(raw) > 2:
            continue
        if aliases.is_alias_name(raw):
            return raw, argv[:index] + argv[index + 1:]
        break
    return None, argv


def _is_completion(argv):
    """Tab-completion invocations are never written to the log file."""
    words = []
    for raw in argv:
        if raw == "--":
            break
        if not raw.startswith("--"):
            words.append(raw)
    return words[:len(_COMPLETION)] == _COMPLETION


def _last_option(options, name):
    found = None
    for key, value in options:
        if key == name:
            found = value
    return found


class _Invocation:
    def __init__(self, argv, env, stdout, stderr, tree, cwd, stdin, started):
        self.argv = argv
        self.env = env
        self.stdout = stdout
        self.stderr = stderr
        self.tree = tree
        self.cwd = cwd
        self.stdin = stdin
        self.started = started

    def _color(self, setting):
        return color_enabled(setting, self.stdout)

    def start(self, alias):
        self.locations = config_files.Locations(self.env)
        log = logfile.LogFile(logfile.log_path(self.locations.root_dir),
                              enabled=not _is_completion(self.argv))
        log.start_invocation()
        self.out = out = Output(self.stdout, self.stderr, stdin=self.stdin, log=log,
                                start=self.started, color=self._color("auto"))
        try:
            items = rewrites.apply(tokens.lex(self.argv))
            self.runtime, self.items = runtime_config.extract(items, self.env)
        except CliError as exc:
            out.report(exc)
            return exc.exit_code
        runtime_values = self.runtime.values
        out.configure(quiet=self.runtime.quiet(),
                      debug=runtime_values.get("debug", False),
                      color=self._color(runtime_values.get("color", "auto")))
        for warning in self.runtime.warnings:
            out.warning(warning)
        self.cache = file_cache.from_env(self.env, self.locations.root_dir,
                                         warn=out.warning)
        try:
            return self.load_and_execute(alias)
        finally:
            if not _is_completion(self.argv):
                try:
                    self.cache.clean()
                except OSError:
                    pass

    def load_and_execute(self, alias):
        out = self.out
        try:
            self.loaded = config_files.load(self.locations, self.cwd)
            if "color" not in self.runtime.values:
                out.configure(color=self._color(self.loaded.values.get("color", "auto")))
            self.registry = aliases.AliasRegistry(
                self.loaded.aliases, aliases.runtime_aliases(self.env))
            for path in self.loaded.files:
                out.debug(f"Loaded config file: {path}", group="bootstrap")
            return self.execute(alias, ())
        except CliError as exc:
            out.report(exc)
            return exc.exit_code
        except Halt as exc:
            return exc.exit_code

    def execute(self, alias, stack):
        if alias is None:
            return self.invoke({})
        target = self.registry.resolve(alias)
        if not target.is_group():
            return self.invoke(target.connection)
        code = 0
        for member in target.members:
            if member == alias or member in stack:
                continue
            self.out.log(member)
            try:
                result = self.execute(member, stack + (alias,))
            except CliError as exc:
                self.out.report(exc)
                result = exc.exit_code
            code = result or code
        return code

    def invoke(self, connection):
        config = runtime_config.layer(
            self.runtime, self.loaded.values, connection,
            locations=self.locations, loaded=self.loaded, aliases=self.registry)
        self.out.configure(color=self._color(config.color()))
        if config.ssh():
            return remote.run(config.ssh(), self.argv, connection,
                              self.runtime.values, config, self.env, self.out,
                              self.stdout, self.stderr)
        return self.run_local(config)

    def run_local(self, config):
        out = self.out
        if self.tree is None:
            self.tree = build_tree()
        tree = self.tree
        items = self.items
        words = [t.text for t in items if not t.is_option()] or ["help"]
        options = [(t.name, t.value) for t in items if t.is_option()]
        node = None
        try:
            node, remaining = dispatch.resolve(tree, words, config.disabled_commands())
            out.debug(f"Resolved command '{node.path}' with arguments {remaining}",
                      group="bootstrap")
            out.log_only(f"Command: {node.path}")
            if not node.is_runnable():
                out.usage(help_text.composite_usage(node))
                return 0
            remaining = self.stdin_fallback(node.synopsis, remaining, options)
            bound = binding.bind(node.path, node.synopsis, node.defaults(),
                                 remaining, options)
            launcher = process.Launcher(
                self.runtime.values, self.env, self.cwd, out,
                config_path=self.locations.global_config_path)
            result = node.run(CommandContext(node, bound, out, config, tree,
                                             cwd=self.cwd, cache=self.cache,
                                             launcher=launcher))
            return int(result or 0)
        except MissingPositional:
            out.usage(help_text.usage_line(node))
            return 1

    def stdin_fallback(self, synopsis, words, options):
        """Read a missing required positional from STDIN, if any is piped."""
        if len(words) >= synopsis.required_positional_count():
            return words
        raw = self.out.read_stdin()
        if raw is None:
            return words
        fmt = (_last_option(options, "format")
               if "format" in synopsis.option_names() else None)
        return list(words) + [values.read_value(raw, fmt)]


def run(argv, env, stdout, stderr, tree=None, cwd=None, stdin=None):
    """Run one invocation and return its exit code.

    ``stdin`` defaults to ``sys.stdin`` at the time input is needed.
    """
    started = time.monotonic()
    alias, argv = _split_alias(list(argv))
    invocation = _Invocation(argv, env, stdout, stderr, tree,
                             cwd if cwd is not None else os.getcwd(), stdin, started)
    return invocation.start(alias)

class RuntimeValues:
    """Global parameters given on the command line."""

    def __init__(self, values, strict, warnings):
        self.values = dict(values)
        self.strict = strict
        self.warnings = list(warnings)

    def debug(self):
        return bool(self.values.get("debug", False))

    def quiet(self):
        return bool(self.values.get("quiet", False))


class RuntimeConfig:
    """Read-only view of the effective global configuration."""

    def __init__(self, values, strict, locations=None, loaded=None, aliases=None):
        self._values = config_spec.defaults()
        self._values.update(values)
        self._strict = strict
        self._locations = locations
        self._loaded = loaded
        self._aliases = aliases

    def get(self, name, default=None):
        return self._values.get(name, default)

    def values(self):
        return dict(self._values)

    def path(self):
        return self._values.get("path")

    def url(self):
        return self._values.get("url")

    def user(self):
        return self._values.get("user")

    def ssh(self):
        return self._values.get("ssh")

    def color(self):
        return self._values.get("color")

    def debug(self):
        return bool(self._values.get("debug", False))

    def quiet(self):
        return bool(self._values.get("quiet", False))

    def strict(self):
        return self._strict

    def disabled_commands(self):
        entries = self._values.get("disabled_commands") or []
        if not isinstance(entries, list):
            entries = [entries]
        return [" ".join(str(e).split()) for e in entries if e is not None]

    def aliases(self):
        """The alias registry, or ``None`` when none was loaded."""
        return self._aliases

    def global_config_path(self):
        return self._loaded.global_config_path if self._loaded else None

    def project_config_path(self):
        return self._loaded.project_config_path if self._loaded else None

    def packages_dir(self):
        return self._locations.packages_dir if self._locations else None

    def root_dir(self):
        return self._locations.root_dir if self._locations else None

    def as_dict(self):
        return {
            "path": self.path(),
            "url": self.url(),
            "user": self.user(),
            "color": self.color(),
            "debug": self.debug(),
            "quiet": self.quiet(),
            "strict_args_mode": self.strict(),
            "global_config_path": self.global_config_path(),
            "project_config_path": self.project_config_path(),
            "packages_dir": self.packages_dir(),
        }


def extract(items, env):
    """Split global parameters out of ``items``.

    Returns ``(RuntimeValues, remaining_tokens)``. In strict mode only tokens
    before the first positional word are considered global.
    """
    strict = bool(env.get(_STRICT_ENV, ""))
    values = {}
    warnings = []
    remaining = []
    seen_positional = False
    for token in items:
        if not token.is_option():
            seen_positional = True
            remaining.append(token)
            continue
        if strict and seen_positional:
            remaining.append(token)
            continue
        key, value, status = _GLOBALS.interpret(token.name, token.value)
        if status == Synopsis.UNKNOWN:
            remaining.append(token)
        elif status == Synopsis.OK:
            values[key] = value
            if config_spec.is_deprecated(key):
                warnings.append(_messages.deprecated_global_parameter(key))
        else:
            raise CliError(_messages.global_parameter_error(
                key, needs_value=(status == Synopsis.NEEDS_VALUE)))
    return RuntimeValues(values, strict, list(dict.fromkeys(warnings))), remaining


def layer(runtime, file_values=None, alias_values=None, **context):
    """Combine the sources: runtime > alias > config files > defaults."""
    values = {}
    for key, value in (file_values or {}).items():
        param = config_spec.get(key)
        if param is not None and param.file:
            values[key] = value
    for key, value in (alias_values or {}).items():
        if config_spec.get(key) is not None and key not in ("debug", "quiet"):
            values[key] = value
    values.update(runtime.values)
    return RuntimeConfig(values, runtime.strict, **context)
"""Exception types shared by several module interfaces."""


class CliError(Exception):
    """A user-facing failure: the message is reported as an error and the
    process exits with ``exit_code``.

    ``message`` may be a list of lines, which is reported as a multi-line
    error. A ``sensitive`` error contains user-supplied material that must
    never reach the log file; ``log_message`` is an optional redacted form
    that may be logged instead.
    """

    def __init__(self, message, exit_code=1, sensitive=False, log_message=None):
        lines = None
        if isinstance(message, (list, tuple)):
            lines = [str(line) for line in message]
            message = "\n".join(lines)
        super().__init__(message)
        self.message = message
        self.lines = lines
        self.exit_code = exit_code
        self.sensitive = sensitive
        self.log_message = log_message


class MissingPositional(Exception):
    """Raised when a command did not receive all of its required positional
    arguments; the caller is expected to show that command's usage."""


class Halt(Exception):
    """Stops the invocation with ``exit_code`` without reporting anything
    (the cause has already been shown, e.g. by a child process)."""

    def __init__(self, exit_code):
        super().__init__(exit_code)
        self.exit_code = exit_code

"""Filesystem cache for downloads and intermediate data.

Secret: where the cache directory lives, the default time-to-live and size
limit (and the environment variables overriding them), how keys are mapped
to file names, which entries count as expired, and the order in which
entries are evicted when the cache grows too large.
"""

import hashlib
import os
import re
import shutil
import time
from urllib.parse import urlsplit

from . import messages as _messages

_DIR_ENV = "APPCTL_CACHE_DIR"
_TTL_ENV = "APPCTL_CACHE_EXPIRY"
_SIZE_ENV = "APPCTL_CACHE_MAX_SIZE"
_DEFAULT_SUBDIR = ".cache"
DEFAULT_TTL = 15552000          # 180 days
DEFAULT_MAX_SIZE = 314572800    # 300 MiB
_UNSAFE = re.compile(r"[^A-Za-z0-9._-]")
_MAX_PART = 200
_HASH_LENGTH = 12


def cache_dir(env, root_dir):
    """``APPCTL_CACHE_DIR`` when set, otherwise ``<root_dir>/.cache/``."""
    configured = env.get(_DIR_ENV) or ""
    if configured:
        return os.path.expanduser(configured)
    return os.path.join(root_dir, _DEFAULT_SUBDIR) + os.sep


def _int_env(env, name, default):
    try:
        value = int(str(env.get(name, "")).strip())
    except ValueError:
        return default
    return value if value >= 0 else default


def from_env(env, root_dir, warn=None):
    """The cache configured by ``env`` (TTL and size may be overridden)."""
    return FileCache(cache_dir(env, root_dir),
                     ttl=_int_env(env, _TTL_ENV, DEFAULT_TTL),
                     max_size=_int_env(env, _SIZE_ENV, DEFAULT_MAX_SIZE),
                     warn=warn)


def _digest(text):
    return hashlib.sha1(text.encode("utf-8", "surrogatepass")).hexdigest()[:_HASH_LENGTH]


def _safe_part(part):
    """A filesystem-safe name for one key segment. Segments that are already
    safe are kept as they are; any other segment gets a digest of its
    original text appended so that distinct segments stay distinct."""
    cleaned = _UNSAFE.sub("-", part)
    if cleaned == part and part not in ("", ".", "..") and len(part) <= _MAX_PART:
        return part
    cleaned = cleaned.strip(".")[:_MAX_PART] or "-"
    return f"{cleaned}~{_digest(part)}"


def sanitize_key(key):
    """Deterministic, filesystem-safe relative path for cache ``key``.

    URLs map to ``misc/<scheme>-<host>[-<port>]/<path>[-<query>]``; other keys
    are split into directories on ``/`` and ``\\``.
    """
    key = str(key)
    url = urlsplit(key) if "://" in key else None
    if url is not None and url.scheme and url.netloc:
        host = f"{url.scheme}-{url.hostname or url.netloc}"
        if url.port:
            host += f"-{url.port}"
        tail = url.path.lstrip("/") + (f"-{url.query}" if url.query else "")
        parts = ["misc", host, tail or "-"]
    else:
        parts = [p for p in key.replace("\\", "/").split("/") if p != ""]
        if not parts:
            parts = [key]
    return "/".join(_safe_part(p) for p in parts)


class FileCache:
    """Key/value store backed by files below a root directory.

    An entry expires ``ttl`` seconds after it was last written (0 disables
    expiry). ``clean`` removes expired entries and then evicts the least
    recently used ones until the total size is at most ``max_size`` (0
    disables the limit).
    """

    def __init__(self, root, ttl=DEFAULT_TTL, max_size=DEFAULT_MAX_SIZE, warn=None):
        self.root = str(root)
        self.ttl = int(ttl)
        self.max_size = int(max_size)
        self._warn = warn
        self.enabled = self._ensure_dir(self.root)

    # --- helpers ----------------------------------------------------------

    def _warning(self, text):
        if self._warn is not None:
            self._warn(text)

    def _ensure_dir(self, directory, report=True):
        if os.path.isdir(directory):
            return True
        try:
            os.makedirs(directory, exist_ok=True)
        except OSError as exc:
            if report:
                self._warning(_messages.directory_uncreatable(directory, exc))
            return False
        return True

    def is_enabled(self):
        return self.enabled

    def get_root(self):
        return self.root

    def filename(self, key):
        return os.path.join(self.root, *sanitize_key(key).split("/"))

    def _expired(self, mtime, ttl, now):
        return ttl > 0 and now - mtime > ttl

    def _effective_ttl(self, ttl):
        return self.ttl if ttl is None else int(ttl)

    def _entries(self):
        """``(path, size, recency)`` for every file, in a stable order."""
        found = []
        for directory, dirs, files in os.walk(self.root):
            dirs.sort()
            for name in sorted(files):
                path = os.path.join(directory, name)
                try:
                    info = os.lstat(path)
                except OSError:
                    continue
                found.append((path, info.st_size, info.st_mtime,
                              max(info.st_atime, info.st_mtime)))
        return found

    def _unlink(self, path):
        try:
            os.unlink(path)
            return True
        except OSError:
            return False

    def _prune_empty_dirs(self):
        for directory, _, _ in sorted(os.walk(self.root), reverse=True):
            if os.path.normpath(directory) == os.path.normpath(self.root):
                continue
            try:
                os.rmdir(directory)
            except OSError:
                pass

    # --- reading ----------------------------------------------------------

    def has(self, key, ttl=None):
        """The entry's file name when it exists and has not expired, else
        ``False``. An expired entry is deleted."""
        if not self.enabled:
            return False
        path = self.filename(key)
        try:
            mtime = os.stat(path).st_mtime
        except OSError:
            return False
        if not os.path.isfile(path):
            return False
        if self._expired(mtime, self._effective_ttl(ttl), time.time()):
            self._unlink(path)
            return False
        return path

    def read(self, key, ttl=None, binary=False):
        """Cached content for ``key``, or ``False`` on a cache miss."""
        path = self.has(key, ttl)
        if not path:
            return False
        try:
            with open(path, "rb") as handle:
                data = handle.read()
            mtime = os.stat(path).st_mtime
            os.utime(path, (time.time(), mtime))   # mark as recently used
        except OSError:
            return False
        if binary:
            return data
        return data.decode("utf-8", errors="surrogateescape")

    def export(self, key, target, ttl=None):
        """Copy the cached entry to ``target``; ``False`` on a cache miss."""
        path = self.has(key, ttl)
        if not path:
            return False
        try:
            directory = os.path.dirname(os.path.abspath(target))
            os.makedirs(directory, exist_ok=True)
            shutil.copyfile(path, target)
            mtime = os.stat(path).st_mtime
            os.utime(path, (time.time(), mtime))
        except OSError:
            return False
        return True

    # --- writing ----------------------------------------------------------

    def is_writable(self, key):
        if not self.enabled:
            return False
        path = self.filename(key)
        if os.path.exists(path):
            return os.path.isfile(path) and os.access(path, os.W_OK)
        directory = os.path.dirname(path)
        while not os.path.exists(directory):
            directory = os.path.dirname(directory)
        return os.path.isdir(directory) and os.access(directory, os.W_OK)

    def _prepare(self, key):
        if not self.enabled:
            return None
        path = self.filename(key)
        if not self._ensure_dir(os.path.dirname(path)):
            return None
        return path

    def _touch(self, path):
        now = time.time()
        os.utime(path, (now, now))

    def write(self, key, contents):
        """Store ``contents`` (text or bytes) under ``key``; return success."""
        path = self._prepare(key)
        if path is None:
            return False
        data = contents if isinstance(contents, bytes) else str(contents).encode(
            "utf-8", errors="surrogateescape")
        try:
            with open(path, "wb") as handle:
                handle.write(data)
            self._touch(path)
        except OSError:
            return False
        return True

    def import_file(self, key, source):
        """Store a copy of the file ``source`` under ``key``; return success."""
        path = self._prepare(key)
        if path is None or not os.path.isfile(source):
            return False
        try:
            shutil.copyfile(source, path)
            self._touch(path)
        except OSError:
            return False
        return True

    def remove(self, key):
        if not self.enabled:
            return False
        path = self.filename(key)
        return os.path.isfile(path) and self._unlink(path)

    # --- maintenance --------------------------------------------------------

    def clean(self, now=None):
        """Delete expired entries, then evict least recently used entries
        (ties broken by path) until the cache fits within ``max_size``."""
        if not self.enabled or not os.path.isdir(self.root):
            return False
        now = time.time() if now is None else now
        kept = []
        for path, size, mtime, recency in self._entries():
            if self._expired(mtime, self.ttl, now):
                self._unlink(path)
            else:
                kept.append((path, size, recency))
        if self.max_size > 0:
            total = 0
            for path, size, _ in sorted(kept, key=lambda e: (-e[2], e[0])):
                if total + size <= self.max_size:
                    total += size
                else:
                    self._unlink(path)
        self._prune_empty_dirs()
        return True

    def clear(self):
        """Delete every entry."""
        if not self.enabled or not os.path.isdir(self.root):
            return False
        for path, _, _, _ in self._entries():
            self._unlink(path)
        self._prune_empty_dirs()
        return True

    def size(self):
        return sum(size for _, size, _, _ in self._entries()) if self.enabled else 0
"""Invocation log file.

Secret: where the log file lives, the timestamp format of each entry, the
delimiter line written at the start of every invocation, and that write
failures never interrupt the command.
"""

import os
from datetime import datetime, timezone

_RELATIVE_PATH = os.path.join("logs", "appctl.log")
_TIMESTAMP = "%Y-%m-%dT%H:%M:%SZ"
_DELIMITER = "========== appctl invocation (pid {pid}) =========="


def log_path(root_dir):
    return os.path.join(root_dir, _RELATIVE_PATH)


def timestamp(now=None):
    """UTC timestamp in ISO 8601 form, e.g. ``2026-01-02T03:04:05Z``."""
    now = now or datetime.now(timezone.utc)
    return now.astimezone(timezone.utc).strftime(_TIMESTAMP)


class LogFile:
    """Appends timestamped entries to the log file (or does nothing when
    disabled)."""

    def __init__(self, path, enabled=True):
        self.path = path
        self.enabled = bool(enabled and path)

    def _append(self, lines):
        if not self.enabled:
            return
        try:
            directory = os.path.dirname(self.path)
            if directory:
                os.makedirs(directory, exist_ok=True)
            with open(self.path, "a", encoding="utf-8") as handle:
                handle.write("".join(line + "\n" for line in lines))
        except OSError:
            pass

    def start_invocation(self):
        self.write(_DELIMITER.format(pid=os.getpid()))

    def write(self, text):
        """Append ``text``; every line carries the same timestamp."""
        stamp = timestamp()
        self._append(f"[{stamp}] {line}" for line in str(text).split("\n"))


class NullLog(LogFile):
    def __init__(self):
        super().__init__(None, enabled=False)
