"""Shared behaviour for plugin and theme management commands."""

from ..errors import AppctlError
from ._listing import render_list

LIST_FIELDS = ["name", "status", "update", "version"]


class ExtensionCatalog:
    """A simulated collection of installed plugins or themes."""

    def __init__(self, noun, items):
        self.noun = noun
        self.items = items

    def find(self, name):
        return next((i for i in self.items if i["name"] == name), None)

    def targets(self, args, assoc):
        """Resolve the named items, or every item with ``--all``."""
        if assoc.get("all"):
            return [i["name"] for i in self.items]
        if not args:
            raise AppctlError(f"Please specify one or more {self.noun}s, or use --all.")
        return list(args)

    def each_installed(self, ctx, names):
        for name in names:
            if self.find(name) is None:
                ctx.output.warning(f"The '{name}' {self.noun} could not be found.")
            else:
                yield name

    # -- handlers -------------------------------------------------------

    def list(self, ctx, args, assoc):
        render_list(ctx, self.items, assoc, LIST_FIELDS, LIST_FIELDS + ["title"], id_field="name")

    def status(self, ctx, args, assoc):
        if args:
            item = self.find(args[0])
            if item is None:
                raise AppctlError(f"The '{args[0]}' {self.noun} could not be found.")
            ctx.output.line(f"{self.noun.capitalize()} {item['name']} details:")
            ctx.output.line(f"    Name: {item['title']}")
            ctx.output.line(f"    Status: {item['status'].capitalize()}")
            ctx.output.line(f"    Version: {item['version']}")
            return
        ctx.output.line(f"{len(self.items)} installed {self.noun}s:")
        for item in self.items:
            marker = "A" if item["status"] == "active" else "I"
            ctx.output.line(f"  {marker} {item['name']} {item['version']}")

    def install(self, ctx, args, assoc):
        for name in args:
            if self.find(name) and not assoc.get("force"):
                ctx.output.warning(f"{name}: {self.noun.capitalize()} already installed.")
                continue
            ctx.output.log(f"Installing {name} ({assoc['version']})...")
            if assoc.get("activate"):
                ctx.output.log(f"Activating '{name}'...")
        ctx.output.success(f"Installed {len(args)} of {len(args)} {self.noun}s.")

    def delete(self, ctx, args, assoc):
        names = self.targets(args, assoc)
        deleted = list(self.each_installed(ctx, names))
        for name in deleted:
            ctx.output.log(f"Deleted '{name}' {self.noun}.")
        self.summary(ctx, "Deleted", len(deleted), len(names))

    def update(self, ctx, args, assoc):
        names = self.targets(args, assoc)
        found = list(self.each_installed(ctx, names))
        if assoc.get("dry-run"):
            ctx.output.log(f"Available {self.noun} updates: {', '.join(found) or 'none'}")
            return
        self.summary(ctx, "Updated", len(found), len(names))

    def summary(self, ctx, verb, done, total):
        if done == total:
            ctx.output.success(f"{verb} {done} of {total} {self.noun}s.")
        else:
            raise AppctlError(f"Only {verb.lower()} {done} of {total} {self.noun}s.")

from ..errors import AppctlError
from ..formatter import format_item, format_items, format_value, select_fields

RESERVED_KEYS = ("field", "fields", "format")

LIST_OPTIONS = {
    "field": "Prints the value of a single field for each item.",
    "fields": "Limit the output to specific object fields.",
    "format": "Render output in a particular format.",
}


def filter_items(items, assoc, available):
    """Keep items matching every generic ``--<field>=<value>`` filter."""
    filters = {k: v for k, v in assoc.items() if k not in RESERVED_KEYS and k in available}
    return [
        item for item in items
        if all(str(item.get(k)) == str(v) for k, v in filters.items())
    ]


def render_list(ctx, items, assoc, default_fields, available=None, id_field="id"):
    available = list(available or default_fields)
    items = filter_items(items, assoc, available)
    fmt = assoc.get("format", "table")
    field = assoc.get("field")
    if field and field is not True:
        if field not in available:
            raise AppctlError(f"Invalid field: {field}.")
        values = [item.get(field) for item in items]
        if fmt == "json":
            ctx.output.line(format_value(values, "json"))
        else:
            for value in values:
                ctx.output.line(format_value(value))
        return
    fields = select_fields(assoc.get("fields"), default_fields, available)
    ctx.output.line(format_items(items, fields, fmt, id_field=id_field))


def render_record(ctx, item, assoc, available):
    field = assoc.get("field")
    if field and field is not True:
        if field not in available:
            raise AppctlError(f"Invalid field: {field}.")
        fmt = "json" if assoc.get("format") == "json" else "plaintext"
        ctx.output.line(format_value(item.get(field), fmt))
        return
    fields = select_fields(assoc.get("fields"), available, available)
    ctx.output.line(format_item(item, fields, assoc.get("format", "table")))


def require_confirmation(assoc, question):
    if not assoc.get("yes"):
        raise AppctlError(f"{question} Re-run with --yes to confirm.")

def version(ctx, args, assoc):
    ctx.output.line(f"{PROGRAM_NAME} {__version__}")


def info(ctx, args, assoc):
    data = {
        "appctl_version": __version__,
        "python_version": platform.python_version(),
        "os": platform.system(),
        "strict_args_mode": ctx.strict,
        "global_config_path": ctx.config.global_path if ctx.config else None,
        "project_config_path": ctx.config.project_path if ctx.config else None,
        "packages_dir": ctx.config.packages_dir if ctx.config else None,
        "runtime_config": {k: v for k, v in ctx.runtime.items()
                           if v != GLOBAL_PARAMETERS[k].default},
    }
    if assoc["format"] == "json":
        ctx.output.line(json.dumps(data))
        return
    if assoc["format"] != "list":
        raise AppctlError(f"Invalid format: {assoc['format']}. Supported formats: list, json.")
    ctx.output.line(f"appctl version:\t{data['appctl_version']}")
    ctx.output.line(f"Python version:\t{data['python_version']}")
    ctx.output.line(f"OS:\t{data['os']}")
    ctx.output.line(f"Strict args mode:\t{'enabled' if ctx.strict else 'disabled'}")
    ctx.output.line(f"Global config:\t{data['global_config_path'] or ''}")
    ctx.output.line(f"Project config:\t{data['project_config_path'] or ''}")
    ctx.output.line(f"Packages dir:\t{data['packages_dir'] or ''}")
    for key, value in data["runtime_config"].items():
        ctx.output.line(f"--{key}:\t{format_value(value)}")


def param_dump(ctx, args, assoc):
    dump = {}
    for name, param in visible_parameters().items():
        entry = {"synopsis": param.synopsis, "description": param.description,
                 "default": param.default}
        if assoc.get("with-values"):
            entry["current"] = ctx.runtime.get(name)
        dump[name] = entry
    ctx.output.line(format_value(dump, "json"))


def has_command(ctx, args, assoc):
    tokens = " ".join(args).split()
    resolved = try_resolve(ctx.root, tokens)
    found = resolved is not None and not resolved[1]
    return 0 if found else 1

        options={"command-name": "The command to check."},
    )

import os

from ..errors import AppctlError
from ._listing import LIST_OPTIONS, render_list

CONFIG_FILE = "config.php"
TYPES = ("constant", "variable", "all")

ENTRIES = [
    {"name": "DB_NAME", "value": "appctl", "type": "constant"},
    {"name": "DB_USER", "value": "appctl", "type": "constant"},
    {"name": "DB_HOST", "value": "localhost", "type": "constant"},
    {"name": "DEBUG", "value": False, "type": "constant"},
    {"name": "table_prefix", "value": "app_", "type": "variable"},
]


def _check_type(entry_type):
    if entry_type not in TYPES:
        raise AppctlError(f"Invalid type '{entry_type}'. Expected one of: {', '.join(TYPES)}.")


def _find(name, entry_type):
    _check_type(entry_type)
    for entry in ENTRIES:
        if entry["name"] == name and entry_type in ("all", entry["type"]):
            return entry
    return None


def _label(entry_type):
    return "constant or variable" if entry_type == "all" else entry_type


def get(ctx, args, assoc):
    entry = _find(args[0], assoc["type"])
    if entry is None:
        raise AppctlError(
            f"The {_label(assoc['type'])} '{args[0]}' is not defined in the '{CONFIG_FILE}' file."
        )
    value = entry["value"]
    ctx.output.line(str(value).lower() if isinstance(value, bool) else str(value))


def set_(ctx, args, assoc):
    name, value = args
    entry = _find(name, assoc["type"])
    kind = entry["type"] if entry else ("constant" if assoc["type"] == "all" else assoc["type"])
    verb = "Updated" if entry else "Added"
    preposition = "in" if entry else "to"
    raw = " (raw)" if assoc.get("raw") else ""
    ctx.output.success(
        f"{verb} the {kind} '{name}' {preposition} the '{CONFIG_FILE}' file with the{raw} value '{value}'."
    )


def delete(ctx, args, assoc):
    entry = _find(args[0], assoc["type"])
    if entry is None:
        raise AppctlError(
            f"The {_label(assoc['type'])} '{args[0]}' is not defined in the '{CONFIG_FILE}' file."
        )
    ctx.output.success(f"Deleted the {entry['type']} '{args[0]}' from the '{CONFIG_FILE}' file.")


def list_(ctx, args, assoc):
    render_list(ctx, ENTRIES, assoc, ["name", "value", "type"], id_field="name")


def create(ctx, args, assoc):
    details = (
        f"database: {assoc['dbname']}, user: {assoc['dbuser']}, "
        f"host: {assoc['dbhost']}, prefix: {assoc['dbprefix']}, locale: {assoc['locale']}"
    )
    ctx.output.success(f"Generated '{CONFIG_FILE}' file ({details}).")


def path(ctx, args, assoc):
    base = ctx.runtime.get("path") or os.getcwd()
    ctx.output.line(os.path.join(str(base), CONFIG_FILE))


def register(registry):

    {"option_name": "siteurl", "option_value": "https://example.test", "autoload": "yes"},
    {"option_name": "timezone_string", "option_value": "UTC", "autoload": "no"},
]
FIELDS = ["option_name", "option_value", "autoload"]


def _find(key):
    return next((o for o in OPTIONS if o["option_name"] == key), None)


def add(ctx, args, assoc):
    key, value = args
    if _find(key):
        raise AppctlError(f"Could not add option '{key}'. Does it already exist?")
    ctx.output.success(f"Added '{key}' option (autoload: {assoc['autoload']}).")


def delete(ctx, args, assoc):
    for key in args:
        if _find(key):
            ctx.output.success(f"Deleted '{key}' option.")
        else:
            ctx.output.warning(f"Could not delete '{key}' option. Does it exist?")


def get(ctx, args, assoc):
    option = _find(args[0])
    if option is None:
        raise AppctlError(f"Could not get '{args[0]}' option. Does it exist?")
    ctx.output.line(format_value(option["option_value"], assoc["format"]))


def list_(ctx, args, assoc):
    items = OPTIONS
    pattern = assoc.get("search")
    if pattern and pattern is not True:
        items = [o for o in items if fnmatch.fnmatchcase(o["option_name"], pattern)]
    autoload = assoc.get("autoload")
    if autoload in ("on", "off", "yes", "no"):
        wanted = "yes" if autoload in ("on", "yes") else "no"
        items = [o for o in items if o["autoload"] == wanted]
    filters = {k: v for k, v in assoc.items() if k not in ("search", "autoload")}
    render_list(ctx, items, filters, ["option_name", "option_value"], FIELDS, id_field="option_name")


def update(ctx, args, assoc):
    key, value = args
    option = _find(key)
    if option and option["option_value"] == value:
        ctx.output.success(f"Value passed for '{key}' option is unchanged.")
    elif option:
        ctx.output.success(f"Updated '{key}' option.")
    else:
        ctx.output.success(f"Added '{key}' option.")

    name = args[0]
    if not SITE_NAME_RE.match(name):
        raise AppctlError(

        )
    if _find(name):
        raise AppctlError(f"Site '{name}' already exists.")
    if assoc.get("porcelain"):
        ctx.output.line(name)
        return
    title = assoc.get("title") or name

                ctx.output.warning(f"Site '{name}' is already {status}.")
            else:
                ctx.output.success(f"Site '{name}' {status}.")
        return 1 if failed else 0
    return handler

    extra = " Uploaded files were removed." if assoc.get("uploads") else ""

     "display_name": "Arthur Author", "roles": "author"},
]
FIELDS = ["ID", "user_login", "user_email", "display_name", "roles"]
META = {1: {"nickname": "admin", "locale": "en_US"}, 2: {"nickname": "editor"}, 3: {}}
ROLES = ("administrator", "editor", "author", "contributor", "subscriber")


def find_user(identifier):
    """Look a user up by ID, login or email."""
    for user in USERS:
        if identifier in (str(user["ID"]), user["user_login"], user["user_email"]):
            return user
    return None


def require_user(identifier):
    user = find_user(identifier)
    if user is None:
        raise AppctlError(f"Invalid user ID, email or login: '{identifier}'")
    return user


def _check_role(role):
    if role not in ROLES:
        raise AppctlError(f"Role doesn't exist: {role}")


def create(ctx, args, assoc):
    login, email = args
    if find_user(login) or find_user(email):
        raise AppctlError(f"Sorry, that username or email address already exists: {login}")
    _check_role(assoc["role"])
    new_id = max(u["ID"] for u in USERS) + 1
    if assoc.get("porcelain"):
        ctx.output.line(str(new_id))
        return
    ctx.output.success(f"Created user {new_id} ({login}, role: {assoc['role']}).")
    if not assoc.get("user_pass"):
        ctx.output.line("Password: (generated)")


def delete(ctx, args, assoc):
    reassign = assoc.get("reassign")
    if reassign:
        require_user(reassign)
    for identifier in args:
        user = require_user(identifier)
        suffix = f" and reassigned posts to user {reassign}" if reassign else ""
        scope = " from the network" if assoc.get("network") else ""
        ctx.output.success(f"Removed user {user['ID']}{scope}{suffix}.")


def get(ctx, args, assoc):
    render_record(ctx, require_user(args[0]), assoc, FIELDS)


def list_(ctx, args, assoc):

    role = filters.pop("role", None)
    if role:
        filters["roles"] = role
    render_list(ctx, USERS, filters, FIELDS, id_field="ID")

    for identifier in args:
        user = require_user(identifier)
        ctx.output.success(f"Updated user {user['ID']}.")


def set_role(ctx, args, assoc):
    user = require_user(args[0])
    role = args[1]
    _check_role(role)
    ctx.output.success(f"Added {user['user_login']} ({user['ID']}) to {role} role.")

    user = require_user(args[0])

def meta_list(ctx, args, assoc):
    user = require_user(args[0])
    items = [{"user_id": user["ID"], "meta_key": k, "meta_value": v}
             for k, v in sorted(META.get(user["ID"], {}).items())]
    keys = assoc.get("keys")
    if keys and keys is not True:
        wanted = set(keys.split(","))
        items = [i for i in items if i["meta_key"] in wanted]
    filters = {k: v for k, v in assoc.items() if k != "keys"}
    render_list(ctx, items, filters, ["user_id", "meta_key", "meta_value"], id_field="meta_key")

    user = require_user(args[0])

    if current == args[2]:

    else:
        ctx.output.success(f"Updated custom field '{args[1]}'.")
