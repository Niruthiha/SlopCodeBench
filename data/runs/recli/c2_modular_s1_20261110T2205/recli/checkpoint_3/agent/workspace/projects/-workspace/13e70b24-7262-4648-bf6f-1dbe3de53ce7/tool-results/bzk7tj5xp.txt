    21	def extract_alias(argv):
    22	    """Return ``(alias, argv_without_alias)``; the alias is the first non-flag token."""
    23	    for index, token in enumerate(argv):
    24	        if token == END_OF_OPTIONS:
    25	            break
    26	        if is_assoc_token(token):
    27	            continue
    28	        if config_loader.is_alias(token):
    29	            return token, argv[:index] + argv[index + 1:]
    30	        break
    31	    return None, list(argv)

    82	    def run_alias(self, name, stack=()):
    83	        aliases = self.loaded.aliases
    84	        if name == config_loader.ALL_ALIAS and not aliases:
    85	            raise AppctlError("Cannot use '@all' when no aliases are registered.")
    86	        if name not in aliases:
    87	            message = f"Alias '{name}' not found."
    88	            suggestion = get_suggestion(name, list(aliases))
    89	            if suggestion:
    90	                message += f"\nDid you mean '{suggestion}'?"
    91	            raise AppctlError(message)
    92	        definition = aliases[name]
    93	        if name == config_loader.ALL_ALIAS and definition is None:
    94	            members = sorted((a for a, d in aliases.items() if a != name and not isinstance(d, list)),
    95	                             key=config_loader.alias_sort_key)
    96	            return self.run_group(name, members, stack)
    97	        if isinstance(definition, list):
    98	            members = _unique(definition)
    99	            invalid = [m for m in members if m not in aliases]
   100	            if invalid:
   101	                raise AppctlError(
   102	                    f"Group '{name}' contains one or more invalid aliases: {', '.join(invalid)}."
   103	                )
   104	            return self.run_group(name, members, stack)
   105	        return self.run(name, definition if isinstance(definition, dict) else {})
   106	
   107	    def run_group(self, name, members, stack):
   108	        stack = stack + (name,)
   109	        result = 0
   110	        for member in members:
   111	            if member in stack:
   112	                continue
   113	            self.output.log(member)
   114	            self.output.stdout.flush()
   115	            code = guarded(self.output, self.run_alias, member, stack)
   116	            self.output.stdout.flush()
   117	            if code:
   118	                result = code
   119	        return result

   121	    # -- a single execution ----------------------------------------------
   122	
   123	    def run(self, alias=None, alias_params=None):
   124	        runtime = build_runtime_config(self.parsed.runtime, self.loaded.merged, alias_params)
   125	        if runtime.get("ssh"):
   126	            return run_remote(runtime["ssh"], self.argv, self.environ, self.output,
   127	                              alias=alias, alias_params=alias_params,
   128	                              skip_tty=bool(runtime.get("skip-tty")))
   129	        positional = self.parsed.positional
   130	        if not positional:
   131	            self.output.line(render_root_help(self.root))
   132	            return 0
   133	        command, remaining = resolve(self.root, positional)
   134	        check_disabled(command, positional, runtime["disabled_commands"])
   135	        if not command.is_runnable:
   136	            self.output.line(render_usage(command))
   137	            return 0
   138	        self.output.debug(f"Running command: {command.full_name}")
   139	        args, assoc = ArgumentValidator(command).validate(
   140	            remaining, self.parsed.assoc, warn=self.output.warning
   141	        )
   142	        context = Context(root=self.root, command=command, output=self.output,
   143	                          runtime=runtime, strict=self.strict, config=self.loaded,
   144	                          alias=alias)
   145	        return command.handler(context, args, assoc)

   148	def main(argv=None, environ=None, stdout=None, stderr=None, cwd=None):
   149	    argv = sys.argv[1:] if argv is None else list(argv)
   150	    environ = os.environ if environ is None else environ

    10	def is_assoc_token(token):
    11	    """True for ``--key``, ``--key=value`` and ``--no-key`` tokens."""
    12	    return token.startswith("--") and len(token) > 2
    13	
    14	
    15	def parse_assoc_token(token):
    16	    """Return ``(key, value)`` for an associative token.
    17	
    18	    ``--key=value`` yields the string value, ``--no-key`` yields ``False``
    19	    and a bare ``--key`` yields ``True``.
    20	    """
    21	    body = token[2:]
    22	    if "=" in body:
    23	        key, value = body.split("=", 1)
    24	        return key, value
    25	    if body.startswith("no-"):
    26	        return body[3:], False
    27	    return body, True

    33	    assoc: dict = field(default_factory=dict)

    37	def parse_argv(argv, strict=False):
    38	    """Classify every argv token.
    39	
    40	    Global parameters become runtime configuration. In strict mode only
    41	    those appearing before the first positional token are treated as
    42	    global; afterwards every argument is command-specific.
    43	    """
    44	    parsed = ParsedArguments()
    45	    end_of_options = False
    46	    seen_positional = False
    47	    for token in argv:
    48	        if not end_of_options and token == END_OF_OPTIONS:
    49	            end_of_options = True
    50	            continue
    51	        if end_of_options or not is_assoc_token(token):
    52	            parsed.positional.append(token)
    53	            seen_positional = True
    54	            continue
    55	        key, value = parse_assoc_token(token)
    56	        if is_global_parameter(key) and not (strict and seen_positional):
    57	            parsed.runtime[key] = value
    58	        else:
    59	            parsed.assoc[key] = value
    60	    return parsed

    12	class Command:
    13	    """A node in the command tree.
    14	
    15	    A command may have a handler, subcommands, or both. A command with
    16	    subcommands but no handler only prints its usage when invoked.
    17	    """
    18	
    19	    def __init__(self, name, description="", synopsis="", handler=None,
    20	                 alias=None, defaults=None, options=None, longdesc=""):
    21	        self.name = name
    22	        self.description = description
    23	        self.synopsis = " ".join((synopsis or "").split())
    24	        self.handler = handler
    25	        self.alias = alias
    26	        self.defaults = dict(defaults or {})
    27	        self.options = dict(options or {})
    28	        self.longdesc = longdesc
    29	        self.parent = None
    30	        self.subcommands = {}
    31	
    32	        self.tokens = parse_synopsis(self.synopsis)
    33	        bad = invalid_tokens(self.tokens)
    34	        if bad:
    35	            raise SynopsisError(
    36	                f"The '{name}' command has an invalid synopsis part: {', '.join(bad)}"
    37	            )
    38	
    39	    # -- tree structure -------------------------------------------------
    40	
    41	    def add_subcommand(self, command):
    42	        for taken in self._taken_names():
    43	            if taken in (command.name, command.alias):
    44	                raise ValueError(
    45	                    f"'{taken}' is already registered under '{self.full_name or 'appctl'}'"
    46	                )
    47	        command.parent = self
    48	        self.subcommands[command.name] = command
    49	        return command
    50	
    51	    def _taken_names(self):
    52	        for sub in self.subcommands.values():
    53	            yield sub.name
    54	            if sub.alias:
    55	                yield sub.alias
    56	
    57	    def find_subcommand(self, token):
    58	        """Match a subcommand by canonical name first, then by alias."""
    59	        if token in self.subcommands:
    60	            return self.subcommands[token]
    61	        for sub in self.subcommands.values():
    62	            if sub.alias and sub.alias == token:
    63	                return sub
    64	        return None
    65	
    66	    def sorted_subcommands(self):
    67	        return [self.subcommands[k] for k in sorted(self.subcommands, key=sort_key)]
    68	
    69	    def subcommand_names(self):
    70	        return list(self.subcommands)
    71	
    72	    # -- properties -----------------------------------------------------
    73	
    74	    @property
    75	    def is_root(self):
    76	        return self.parent is None
    77	
    78	    @property
    79	    def has_subcommands(self):
    80	        return bool(self.subcommands)
    81	
    82	    @property
    83	    def is_runnable(self):
    84	        return self.handler is not None
    85	
    86	    @property
    87	    def path(self):
    88	        """Canonical names from the top-level command down to this one."""
    89	        parts = []
    90	        node = self
    91	        while node is not None and not node.is_root:
    92	            parts.append(node.name)
    93	            node = node.parent
    94	        return list(reversed(parts))

    98	        return " ".join(self.path)

   101	        return [t for t in self.tokens if t.type in types]

    75	def _read_mapping(path):
    76	    with open(path, encoding="utf-8") as handle:
    77	        text = handle.read()
    78	    try:
    79	        data = yaml.safe_load(text)
    80	    except yaml.YAMLError as exc:

    82	    if data is None:

    89	def ensure_global_config(path, strict=True):
    90	    """Add missing required keys to the global config file, creating it.
    91	
    92	    Write failures raise :class:`AppctlError` unless ``strict`` is false.
    93	    """
    94	    try:
    95	        if os.path.exists(path):
    96	            data, text = _read_mapping(path)
    97	        else:
    98	            data, text = {}, ""
    99	        missing = [(k, v) for k, v in REQUIRED_GLOBAL_KEYS if k not in data]
   100	        if not missing:
   101	            return
   102	        if text and not text.endswith("\n"):
   103	            text += "\n"
   104	        text += "".join(f"{key}: {value}\n" for key, value in missing)
   105	        directory = os.path.dirname(path)
   106	        if directory:
   107	            os.makedirs(directory, exist_ok=True)
   108	        with open(path, "w", encoding="utf-8") as handle:
   109	            handle.write(text)
   110	    except OSError as exc:
   111	        if strict:
   112	            reason = exc.strerror or str(exc)
   113	            raise AppctlError(f"Unable to write the config file '{path}': {reason}.") from exc

   119	def _merge(base, override, append_arrays):
   120	    """Overlay ``override`` on ``base``; nested mappings merge recursively."""
   121	    if isinstance(base, dict) and isinstance(override, dict):
   122	        merged = dict(base)
   123	        for key, value in override.items():
   124	            merged[key] = _merge(base[key], value, append_arrays) if key in base else value
   125	        return merged
   126	    if append_arrays and isinstance(base, list) and isinstance(override, list):
   127	        return base + override
   128	    return override

   132	    for key, value in data.items():
   133	        if is_alias(key) and isinstance(value, dict):
   134	            path = value.get("path")
   135	            if isinstance(path, str) and path and "ssh" not in value:
   136	                value["path"] = os.path.normpath(os.path.join(directory, os.path.expanduser(path)))

   139	def load_config_file(path, accumulated=None, _seen=None):
   140	    """Merge the YAML file at ``path`` (and its ancestors) over ``accumulated``."""
   141	    accumulated = dict(accumulated or {})
   142	    if not path:
   143	        return accumulated
   144	    path = os.path.abspath(path)
   145	    seen = set(_seen or ())
   146	    if path in seen or not os.path.isfile(path):
   147	        return accumulated
   148	    seen.add(path)
   149	    try:
   150	        data, _ = _read_mapping(path)
   151	    except OSError:
   152	        return accumulated
   153	    directory = os.path.dirname(path)
   154	    meta = data.pop(META_KEY, None)
   155	    meta = meta if isinstance(meta, dict) else {}
   156	    inherit = meta.get("inherit")
   157	    if isinstance(inherit, str) and inherit:
   158	        base = os.path.join(directory, os.path.expanduser(inherit))
   159	        accumulated = load_config_file(base, accumulated, seen)
   160	    _absolutize_alias_paths(data, directory)
   161	    return _merge(accumulated, data, meta.get("merge") is True)

   167	def runtime_aliases(environ, cwd):
   168	    raw = environ.get(RUNTIME_ALIAS_ENV)
   169	    if not raw:
   170	        return {}
   171	    try:
   172	        decoded = json.loads(raw)
   173	    except ValueError:
   174	        return {}
   175	    if not isinstance(decoded, dict):
   176	        return {}
   177	    aliases = {k: v for k, v in decoded.items() if is_alias(k)}
   178	    _absolutize_alias_paths(aliases, cwd)
   179	    return aliases

   182	def collect_aliases(config, extra=None):
   183	    """Return ``{name: definition}``, registering ``@all`` when needed.
   184	
   185	    A definition is either a mapping of connection parameters or a list
   186	    of alias names (a group).
   187	    """
   188	    aliases = {}
   189	    for key, value in list(config.items()) + list((extra or {}).items()):
   190	        if not is_alias(key):
   191	            continue
   192	        if isinstance(value, list):
   193	            aliases[key] = [str(v) for v in value]
   194	        elif isinstance(value, dict):
   195	            aliases[key] = dict(value)
   196	        elif value is None:
   197	            aliases[key] = {}
   198	        else:
   199	            aliases[key] = value
   200	    if aliases and ALL_ALIAS not in aliases:
   201	        aliases[ALL_ALIAS] = None  # implicit: every registered alias
   202	    return aliases

    69	    scheme = bits.get("scheme", "ssh")
    70	    host = bits["host"]
    71	    user = bits.get("user")

    73	        argv = ["docker", "exec"]
    74	        if user:
    75	            argv += ["--user", user]
    76	        argv += ["-it" if tty else "-i", host, "sh", "-c", remote_command]
    77	        return argv

    79	        argv = ["docker-compose", "exec" if scheme == "docker-compose" else "run"]

    30	def resolve(root, tokens):
    31	    """Walk the tree and return ``(command, remaining_tokens)``.
    32	
    33	    Tokens keep matching subcommands (by name or alias) until the deepest
    34	    command is found. If the next token does not match, a command with a
    35	    handler receives it as an argument; otherwise it is an error.
    36	    """
    37	    command = root
    38	    index = 0
    39	    while index < len(tokens) and command.has_subcommands:
    40	        child = command.find_subcommand(tokens[index])
    41	        if child is None:
    42	            if command.is_runnable and not command.is_root:
    43	                break
    44	            raise not_found_error(tokens[index], command)
    45	        command = child
    46	        index += 1
    47	    return command, list(tokens[index:])

    32	def levenshtein(source, target):
    33	    """Return the edit distance between two strings."""
    34	    if source == target:
    35	        return 0
    36	    if not source:
    37	        return len(target)
    38	    if not target:
    39	        return len(source)
    40	    previous = list(range(len(target) + 1))
    41	    for i, s_char in enumerate(source, start=1):
    42	        current = [i]
    43	        for j, t_char in enumerate(target, start=1):
    44	            current.append(min(
    45	                previous[j] + 1,
    46	                current[j - 1] + 1,
    47	                previous[j - 1] + (s_char != t_char),
    48	            ))
    49	        previous = current
    50	    return previous[-1]
    51	
    52	
    53	def get_suggestion(target, options, threshold=DEFAULT_THRESHOLD):
    54	    """Return the closest option to ``target`` or ``None``.
    55	
    56	    Synonyms win over edit distance. A match must be within ``threshold``
    57	    edits and must not replace every character of ``target``. Ties are
    58	    broken alphabetically (case-insensitively) so the result is stable.
    59	    """
    60	    candidates = sorted(set(options), key=lambda o: (o.lower(), o))
    61	    if not candidates:
    62	        return None
    63	
    64	    synonym = SYNONYMS.get(target)
    65	    if synonym is not None and synonym in candidates:
    66	        return synonym
    67	
    68	    best_distance, best = None, None
    69	    for option in candidates:
    70	        distance = levenshtein(target, option)
    71	        if best_distance is None or distance < best_distance:
    72	            best_distance, best = distance, option
    73	
    74	    if best_distance <= threshold and best_distance < len(target) and best != target:
    75	        return best
    76	    return None

     1	"""Parsing and classification of command synopsis strings."""
     2	
     3	import re
     4	from dataclasses import dataclass
     5	from typing import Optional
     6	
     7	POSITIONAL = "positional"
     8	ASSOC = "assoc"
     9	FLAG = "flag"
    10	GENERIC = "generic"
    11	UNKNOWN = "unknown"
    12	
    13	GENERIC_TOKEN = "--<field>=<value>"
    14	
    15	_NAME = r"[a-z0-9_-]+"
    16	_VALUE = r"[a-zA-Z0-9_|,-]+"
    17	_POSITIONAL_RE = re.compile(rf"^<({_VALUE})>$")
    18	_PARAM_RE = re.compile(rf"^--(\[no-\])?({_NAME})")
    19	_VALUE_RE = re.compile(rf"^=<({_VALUE})>$")
    20	
    21	
    22	@dataclass(frozen=True)
    23	class SynopsisToken:
    24	    """One classified token of a synopsis string."""
    25	
    26	    token: str
    27	    type: str
    28	    name: Optional[str] = None
    29	    optional: bool = False
    30	    repeating: bool = False
    31	    value_name: Optional[str] = None
    32	    value_optional: bool = False
    33	    negatable: bool = False
    34	
    35	    @property
    36	    def is_positional(self):
    37	        return self.type == POSITIONAL
    38	
    39	    @property
    40	    def is_parameter(self):
    41	        """True for named ``--key=<value>`` and ``--flag`` tokens."""
    42	        return self.type in (ASSOC, FLAG)
    43	
    44	
    45	def _strip_optional(text):
    46	    if len(text) >= 2 and text.startswith("[") and text.endswith("]"):
    47	        return True, text[1:-1]
    48	    return False, text
    49	
    50	
    51	def _strip_repeating(text):
    52	    if text.endswith("..."):
    53	        return True, text[:-3]
    54	    return False, text
    55	
    56	
    57	def classify_token(token):
    58	    """Classify a single synopsis token."""
    59	    optional, body = _strip_optional(token)
    60	    repeating, body = _strip_repeating(body)
    61	    unknown = SynopsisToken(token=token, type=UNKNOWN)
    62	
    63	    if body == GENERIC_TOKEN:
    64	        return SynopsisToken(token, GENERIC, optional=optional, repeating=repeating)
    65	
    66	    match = _POSITIONAL_RE.match(body)
    67	    if match:
    68	        return SynopsisToken(token, POSITIONAL, name=match.group(1),
    69	                             optional=optional, repeating=repeating)
    70	
    71	    match = _PARAM_RE.match(body)
    72	    if not match:
    73	        return unknown
    74	    negatable = bool(match.group(1))
    75	    name = match.group(2)
    76	    rest = body[match.end():]
    77	
    78	    if rest == "":
    79	        # Boolean flags are by nature optional.
    80	        if not optional:
    81	            return unknown
    82	        return SynopsisToken(token, FLAG, name=name, optional=True,
    83	                             repeating=repeating, negatable=negatable)
    84	
    85	    if negatable:
    86	        return unknown
    87	    value_optional, rest = _strip_optional(rest)
    88	    value_match = _VALUE_RE.match(rest)
    89	    if not value_match:
    90	        return unknown
    91	    return SynopsisToken(token, ASSOC, name=name, optional=optional,
    92	                         repeating=repeating, value_name=value_match.group(1),
    93	                         value_optional=value_optional)
    94	
    95	
    96	def parse_synopsis(synopsis):
    97	    """Split a synopsis string into classified tokens."""
    98	    return [classify_token(token) for token in (synopsis or "").split()]
    99	
   100	
   101	def invalid_tokens(tokens):
   102	    """Return the raw text of every malformed token."""
   103	    return [t.token for t in tokens if t.type == UNKNOWN]

     1	"""Validation of supplied arguments against a command's synopsis."""
     2	
     3	from .errors import AppctlError, UsageError
     4	from .help import render_usage
     5	from .suggest import get_suggestion
     6	from .synopsis import ASSOC, FLAG, GENERIC, POSITIONAL
     7	
     8	
     9	class ArgumentValidator:
    10	    """Checks positional and associative arguments for one command."""
    11	
    12	    def __init__(self, command):
    13	        self.command = command
    14	        self.positionals = command.tokens_of_type(POSITIONAL)
    15	        self.parameters = command.tokens_of_type(ASSOC, FLAG)
    16	        self.accepts_any_assoc = bool(command.tokens_of_type(GENERIC))
    17	
    18	    # -- positional -----------------------------------------------------
    19	
    20	    def enough_positionals(self, args):
    21	        required = sum(1 for t in self.positionals if not t.optional)
    22	        return len(args) >= required
    23	
    24	    def unknown_positionals(self, args):
    25	        if any(t.repeating for t in self.positionals):
    26	            return []
    27	        return list(args[len(self.positionals):])
    28	
    29	    # -- associative ----------------------------------------------------
    30	
    31	    def assoc_errors(self, assoc, warn):
    32	        """Return fatal errors; drop optional keyed params given no value."""
    33	        errors = []
    34	        for param in self.parameters:
    35	            key = param.name
    36	            if param.type != ASSOC:
    37	                continue
    38	            value = assoc.get(key)
    39	            if not param.optional:
    40	                if value is None or value is True or value is False or value == "":
    41	                    errors.append(f"missing --{key} parameter")
    42	            elif value is True and not param.value_optional:
    43	                warn(f"--{key} parameter needs a value")
    44	                del assoc[key]
    45	        for key in self.unknown_assoc(assoc):
    46	            message = f"unknown --{key} parameter"
    47	            suggestion = get_suggestion(key, [p.name for p in self.parameters])
    48	            if suggestion:
    49	                message += f"\nDid you mean '--{suggestion}'?"
    50	            errors.append(message)
    51	        return errors
    52	
    53	    def unknown_assoc(self, assoc):
    54	        if self.accepts_any_assoc:
    55	            return []
    56	        known = {p.name for p in self.parameters}
    57	        return [key for key in assoc if key not in known]
    58	
    59	    # -- defaults -------------------------------------------------------
    60	
    61	    def apply_defaults(self, args, assoc):
    62	        defaults = self.command.defaults
    63	        for index, token in enumerate(self.positionals):
    64	            if index < len(args):
    65	                continue
    66	            if token.name in defaults and len(args) == index:
    67	                args.append(defaults[token.name])
    68	            else:
    69	                break
    70	        for param in self.parameters:
    71	            if param.name not in assoc and param.name in defaults:
    72	                assoc[param.name] = defaults[param.name]
    73	
    74	    # -- entry point ----------------------------------------------------
    75	
    76	    def validate(self, args, assoc, warn=lambda message: None):
    77	        """Validate and return ``(args, assoc)`` with defaults applied."""
    78	        args = list(args)
    79	        assoc = dict(assoc)
    80	        if not self.enough_positionals(args):
    81	            raise UsageError(render_usage(self.command))
    82	        extra = self.unknown_positionals(args)
    83	        if extra:
    84	            raise AppctlError("Too many positional arguments: " + " ".join(extra))
    85	        errors = self.assoc_errors(assoc, warn)
    86	        if errors:
    87	            raise AppctlError("Parameter errors:\n" + "\n".join(f" {e}" for e in errors))
    88	        self.apply_defaults(args, assoc)
    89	        return args, assoc
