"""Scope and binding model.

Secret: how name bindings are discovered in a syntax tree, which scope they
belong to, and the rules (indentation heuristics included) that decide which
scope encloses a cursor and which bindings are visible from it.
"""

import ast
import re

from . import parsing

# Binding forms exposed to clients.
DEF = "def"
CLASS = "class"
PARAM = "param"
ASSIGN = "assign"
IMPORT = "import"
IMPORT_FROM = "import_from"
STAR = "star"
EXCEPT = "except"
OTHER = "other"


class Binding:
    """One event that binds a name (or, for STAR, a set of names)."""

    def __init__(self, name, form, line, order, scope, node=None, position=None,
                 stmt=None, home=None, **details):
        self.name = name
        self.form = form
        self.line = line
        self.order = order
        self.scope = scope
        # where the bound identifier itself is written: (1-based line, 0-based column)
        self.position = position or (line, 0)
        # the syntax node naming the binding, the statement (or handler/case)
        # that performs it, and the scope whose code contains that statement
        self.node = node
        self.stmt = stmt
        self.home = home or scope
        # form-specific details:
        #   DEF/CLASS: node
        #   PARAM: function (Scope), index, annotation, default, star ("", "*", "**"),
        #          keyword_only (optional)
        #   ASSIGN: value (expr or None), path (tuple of ints), annotation,
        #           owner (class Scope, for instance attributes), starred
        #   OTHER: source ("for"/"with"/None), value (iterable/context expr), path
        #   IMPORT: module (dotted name bound), as_name (bool)
        #   IMPORT_FROM: module, level, attr
        #   STAR: module, level
        #   EXCEPT: type_expr
        self.details = details


class Scope:
    MODULE, FUNCTION, CLASS = "module", "function", "class"

    def __init__(self, kind, node, parent, tree):
        self.kind = kind
        self.node = node
        self.parent = parent
        self.tree = tree
        self.children = []
        self.events = []
        self.instance_events = []  # self.<attr> assignments in methods (classes)
        self.globals_declared = set()

    @property
    def name(self):
        return getattr(self.node, "name", None)

    def class_scope(self):
        """For a method, the scope of the class that defines it."""
        if self.kind == Scope.FUNCTION and self.parent is not None \
                and self.parent.kind == Scope.CLASS:
            return self.parent
        return None

    def decorator_names(self):
        names = []
        for deco in getattr(self.node, "decorator_list", []):
            if isinstance(deco, ast.Name):
                names.append(deco.id)
            elif isinstance(deco, ast.Attribute):
                names.append(deco.attr)
        return names

    def returned_expressions(self):
        """Return-value expressions of a function (excluding nested scopes)."""
        found = []
        for node in _walk_own(self.node.body if self.node is not None else []):
            if isinstance(node, (ast.Yield, ast.YieldFrom)):
                return []
            if isinstance(node, ast.Return) and node.value is not None:
                found.append(node.value)
        return found

    def return_statements(self):
        """``return`` statements of a function, or None for generators."""
        found = []
        for node in _walk_own(self.node.body if self.node is not None else []):
            if isinstance(node, (ast.Yield, ast.YieldFrom)):
                return None
            if isinstance(node, ast.Return):
                found.append(node)
        return found

    def base_expressions(self):
        return list(getattr(self.node, "bases", []))

    def is_async(self):
        return isinstance(self.node, ast.AsyncFunctionDef)

    def chain(self):
        """Scopes searched for names from here: self, enclosing functions, module."""
        result = [self]
        scope = self.parent
        while scope is not None:
            if scope.kind != Scope.CLASS:
                result.append(scope)
            scope = scope.parent
        return result

    def events_until_line(self, line):
        return [e for e in self.events if e.line <= line]

    def events_before(self, order):
        return [e for e in self.events if e.order < order]

    def all_events(self):
        return list(self.events)

    def instance_attribute_events(self):
        return list(self.instance_events)


class ScopeTree:
    def __init__(self, text, path=None):
        self.path = path
        self._lines = text.split("\n")
        module = parsing.parse_tolerant(text)
        self.module = Scope(Scope.MODULE, module, None, self)
        _Collector(self.module).visit_body(module.body)

    def line_text(self, line):
        return self._lines[line - 1] if 0 < line <= len(self._lines) else ""

    def segment(self, node):
        """Source text of ``node`` (as written), or None."""
        try:
            first, last = node.lineno, node.end_lineno
            if last is None or first > len(self._lines):
                return None
            if first == last:
                return self._lines[first - 1][node.col_offset:node.end_col_offset]
            parts = [self._lines[first - 1][node.col_offset:]]
            parts.extend(self._lines[first:last - 1])
            parts.append(self._lines[last - 1][:node.end_col_offset])
            return "\n".join(parts)
        except (AttributeError, IndexError, TypeError):
            return None

    def find_identifier(self, name, line, col, after_keyword=None):
        """Position of identifier ``name`` at or after (line, col).

        With ``after_keyword`` the identifier must follow that keyword.
        """
        pattern = re.compile((r"\b%s\s+" % after_keyword if after_keyword else r"(?<![\w.])")
                             + r"(%s)\b" % re.escape(name))
        for number in range(line, min(line + 50, len(self._lines)) + 1):
            text = self._lines[number - 1]
            start = col if number == line else 0
            match = pattern.search(text, start)
            if match:
                return number, match.start(1)
        return line, col

    def definition_position(self, node):
        """Position of the name in a ``def``/``class`` header."""
        keyword = "class" if isinstance(node, ast.ClassDef) else "def"
        return self.find_identifier(node.name, node.lineno, node.col_offset, keyword)

    def scope_at(self, line, col):
        scope = self.module
        while True:
            inner = None
            for child in scope.children:
                if self._contains(child, line, col):
                    inner = child
            if inner is None:
                return scope
            scope = inner

    def _contains(self, scope, line, col):
        node = scope.node
        header, end = node.lineno, node.end_lineno or node.lineno
        header_col = node.col_offset
        body = node.body[0] if node.body else None
        if line < header:
            return False
        if line == header:
            return body is not None and body.lineno == header and col >= body.col_offset
        text = self._lines[line - 1] if line - 1 < len(self._lines) else ""
        before = text[:col]
        indent = _indent_width(text) if before.strip() else col
        if indent <= header_col and (line > end or not before.strip()):
            return False
        if line <= end:
            return True
        for number in range(end + 1, line):
            other = self._lines[number - 1] if number - 1 < len(self._lines) else ""
            stripped = other.strip()
            if stripped and not stripped.startswith("#") and _indent_width(other) <= header_col:
                return False
        return True


def _indent_width(text):
    return len(text) - len(text.lstrip())


def _walk_own(statements):
    """Walk nodes in ``statements`` without entering nested scopes."""
    stack = list(reversed(statements))
    while stack:
        node = stack.pop()
        yield node
        if isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef, ast.ClassDef, ast.Lambda)):
            continue
        stack.extend(reversed(list(ast.iter_child_nodes(node))))


class _Counter:
    value = 0

    @classmethod
    def next(cls):
        cls.value += 1
        return cls.value


class _Collector(ast.NodeVisitor):
    def __init__(self, scope):
        self.scope = scope
        self.stmt = None

    def visit(self, node):
        if isinstance(node, (ast.stmt, ast.ExceptHandler, ast.match_case)):
            saved, self.stmt = self.stmt, node
            try:
                return super().visit(node)
            finally:
                self.stmt = saved
        return super().visit(node)

    # -- helpers ---------------------------------------------------------
    def visit_body(self, statements):
        for statement in statements:
            self.visit(statement)

    def _target_scope(self, name):
        if name in self.scope.globals_declared:
            scope = self.scope
            while scope.parent is not None:
                scope = scope.parent
            return scope
        return self.scope

    def bind(self, name, form, line, node=None, position=None, **details):
        scope = self._target_scope(name) if name else self.scope
        binding = Binding(name, form, line, _Counter.next(), scope, node=node,
                          position=position, stmt=self.stmt, home=self.scope, **details)
        scope.events.append(binding)
        return binding

    def bind_target(self, target, line, value, path=(), annotation=None, starred=False):
        if isinstance(target, ast.Name):
            self.bind(target.id, ASSIGN, line, node=target, position=_start(target),
                      value=value, path=path, annotation=annotation, starred=starred)
        elif isinstance(target, (ast.Tuple, ast.List)):
            for index, element in enumerate(target.elts):
                self.bind_target(element, line, value, path + (index,))
        elif isinstance(target, ast.Starred):
            self.bind_target(target.value, line, None, starred=True)
        else:
            self.visit(target)

    def bind_plain(self, target, line, source=None, value=None, path=()):
        if isinstance(target, ast.Name):
            self.bind(target.id, OTHER, line, node=target, position=_start(target),
                      source=source, value=value, path=path)
        elif isinstance(target, (ast.Tuple, ast.List)):
            for index, element in enumerate(target.elts):
                self.bind_plain(element, line, source, value, path + (index,))
        elif isinstance(target, ast.Starred):
            self.bind_plain(target.value, line)
        else:
            for node in ast.walk(target):
                if isinstance(node, ast.Name) and isinstance(node.ctx, ast.Store):
                    self.bind(node.id, OTHER, line, node=node, position=_start(node))

    # -- scopes ----------------------------------------------------------
    def _visit_function(self, node):
        for deco in node.decorator_list:
            self.visit(deco)
        args = node.args
        for default in args.defaults + [d for d in args.kw_defaults if d is not None]:
            self.visit(default)
        tree = self.scope.tree
        self.bind(node.name, DEF, node.lineno, node=node,
                  position=tree.definition_position(node))
        child = Scope(Scope.FUNCTION, node, self.scope, tree)
        self.scope.children.append(child)
        positional = args.posonlyargs + args.args
        defaults = [None] * (len(positional) - len(args.defaults)) + list(args.defaults)
        index = 0
        for arg, default in zip(positional, defaults):
            child_bind(child, arg, node, function=child, index=index,
                       annotation=arg.annotation, default=default, star="")
            index += 1
        if args.vararg:
            child_bind(child, args.vararg, node, function=child, index=index,
                       annotation=None, default=None, star="*")
            index += 1
        for arg, default in zip(args.kwonlyargs, args.kw_defaults):
            child_bind(child, arg, node, function=child, index=index,
                       annotation=arg.annotation, default=default, star="", keyword_only=True)
            index += 1
        if args.kwarg:
            child_bind(child, args.kwarg, node, function=child, index=index,
                       annotation=None, default=None, star="**")
        collector = _Collector(child)
        for statement in _walk_own(node.body):
            if isinstance(statement, ast.Global):
                child.globals_declared.update(statement.names)
        collector.visit_body(node.body)
        parent_class = child.class_scope()
        if parent_class is not None and positional:
            decorators = child.decorator_names()
            if "staticmethod" not in decorators and "classmethod" not in decorators:
                self_name = positional[0].arg
                _collect_instance_attributes(parent_class, child, node.body, self_name,
                                             initializer=node.name == "__init__")

    visit_FunctionDef = _visit_function
    visit_AsyncFunctionDef = _visit_function

    def visit_ClassDef(self, node):
        for deco in node.decorator_list:
            self.visit(deco)
        for base in node.bases:
            self.visit(base)
        tree = self.scope.tree
        self.bind(node.name, CLASS, node.lineno, node=node,
                  position=tree.definition_position(node))
        child = Scope(Scope.CLASS, node, self.scope, tree)
        self.scope.children.append(child)
        _Collector(child).visit_body(node.body)

    def visit_Lambda(self, node):
        pass

    def _visit_comprehension(self, node):
        for generator in node.generators:
            self.visit(generator.iter)
            break

    visit_ListComp = _visit_comprehension
    visit_SetComp = _visit_comprehension
    visit_DictComp = _visit_comprehension
    visit_GeneratorExp = _visit_comprehension

    # -- statements ------------------------------------------------------
    def visit_Global(self, node):
        self.scope.globals_declared.update(node.names)

    def visit_Assign(self, node):
        self.visit(node.value)
        for target in node.targets:
            self.bind_target(target, node.lineno, node.value)

    def visit_AnnAssign(self, node):
        if node.value is not None:
            self.visit(node.value)
        if isinstance(node.target, ast.Name):
            self.bind(node.target.id, ASSIGN, node.lineno, node=node.target,
                      position=_start(node.target), value=node.value, path=(),
                      annotation=node.annotation)
        else:
            self.visit(node.target)

    def visit_AugAssign(self, node):
        self.visit(node.value)
        if isinstance(node.target, ast.Name):
            name = node.target.id
            scope = self._target_scope(name)
            if not any(e.name == name for e in scope.events):
                self.bind(name, OTHER, node.lineno, node=node.target,
                          position=_start(node.target))
        else:
            self.visit(node.target)

    def visit_NamedExpr(self, node):
        self.visit(node.value)
        if isinstance(node.target, ast.Name):
            self.bind(node.target.id, ASSIGN, node.lineno, node=node.target,
                      position=_start(node.target), value=node.value, path=(),
                      annotation=None)

    def _visit_for(self, node):
        self.visit(node.iter)
        self.bind_plain(node.target, node.lineno, "for", node.iter)
        self.visit_body(node.body)
        self.visit_body(node.orelse)

    visit_For = _visit_for
    visit_AsyncFor = _visit_for

    def _visit_with(self, node):
        for item in node.items:
            self.visit(item.context_expr)
            if item.optional_vars is not None:
                self.bind_plain(item.optional_vars, node.lineno, "with", item.context_expr)
        self.visit_body(node.body)

    visit_With = _visit_with
    visit_AsyncWith = _visit_with

    def visit_ExceptHandler(self, node):
        if node.type is not None:
            self.visit(node.type)
        if node.name:
            tree = self.scope.tree
            position = tree.find_identifier(node.name, node.lineno, node.col_offset, "as")
            self.bind(node.name, EXCEPT, node.lineno, node=node, position=position,
                      type_expr=node.type)
        self.visit_body(node.body)

    def visit_Import(self, node):
        for alias in node.names:
            if alias.asname:
                self.bind(alias.asname, IMPORT, node.lineno, node=alias,
                          position=_as_position(alias), module=alias.name, as_name=True)
            else:
                first = alias.name.split(".")[0]
                self.bind(first, IMPORT, node.lineno, node=alias, position=_start(alias),
                          module=first, as_name=False)

    def visit_ImportFrom(self, node):
        module = node.module or ""
        for alias in node.names:
            if alias.name == "*":
                self.bind(None, STAR, node.lineno, node=alias, module=module, level=node.level)
            else:
                position = _as_position(alias) if alias.asname else _start(alias)
                self.bind(alias.asname or alias.name, IMPORT_FROM, node.lineno, node=alias,
                          position=position, module=module, level=node.level, attr=alias.name)

    def visit_MatchAs(self, node):
        if node.pattern is not None:
            self.visit(node.pattern)
        if node.name:
            position = (node.end_lineno, node.end_col_offset - len(node.name))
            self.bind(node.name, OTHER, node.lineno, node=node, position=position)

    def visit_MatchStar(self, node):
        if node.name:
            position = (node.end_lineno, node.end_col_offset - len(node.name))
            self.bind(node.name, OTHER, node.lineno, node=node, position=position)

    def visit_MatchMapping(self, node):
        self.generic_visit(node)
        if node.rest:
            position = self.scope.tree.find_identifier(node.rest, node.lineno, node.col_offset)
            self.bind(node.rest, OTHER, node.lineno, node=node, position=position)

    def visit_TypeAlias(self, node):
        if isinstance(node.name, ast.Name):
            self.bind(node.name.id, OTHER, node.lineno, node=node.name,
                      position=_start(node.name))


def _start(node):
    return (getattr(node, "lineno", 1), getattr(node, "col_offset", 0))


def _as_position(alias):
    """Position of the ``as`` name of an import alias."""
    return (alias.end_lineno, alias.end_col_offset - len(alias.asname))


def child_bind(scope, arg, function_node, **details):
    scope.events.append(Binding(arg.arg, PARAM, function_node.lineno, _Counter.next(), scope,
                                node=arg, position=_start(arg), stmt=function_node,
                                home=scope, **details))


def _collect_instance_attributes(class_scope, method_scope, body, self_name, initializer):
    """Record ``<self>.<attr> = ...`` assignments of a method on its class."""
    found = []

    def is_self_attr(target):
        return (isinstance(target, ast.Attribute) and isinstance(target.value, ast.Name)
                and target.value.id == self_name)

    def record(target, statement, value, path, annotation=None):
        if is_self_attr(target):
            found.append(Binding(
                target.attr, ASSIGN, statement.lineno, _Counter.next(), method_scope,
                node=target, position=(target.end_lineno,
                                       target.end_col_offset - len(target.attr)),
                stmt=statement, home=method_scope, value=value, path=path,
                annotation=annotation, owner=class_scope))
        elif isinstance(target, (ast.Tuple, ast.List)):
            for index, element in enumerate(target.elts):
                record(element, statement, value, path + (index,))
        elif isinstance(target, ast.Starred):
            record(target.value, statement, None, ())

    for node in _walk_own(body):
        if isinstance(node, ast.Assign):
            for target in node.targets:
                record(target, node, node.value, ())
        elif isinstance(node, ast.AnnAssign):
            record(node.target, node, node.value, (), node.annotation)
        elif isinstance(node, ast.AugAssign) and is_self_attr(node.target):
            if not any(e.name == node.target.attr for e in class_scope.instance_events + found):
                target = node.target
                found.append(Binding(
                    target.attr, OTHER, node.lineno, _Counter.next(), method_scope,
                    node=target, position=(target.end_lineno,
                                           target.end_col_offset - len(target.attr)),
                    stmt=node, home=method_scope, owner=class_scope))
    if initializer:
        class_scope.instance_events[:0] = found
    else:
        class_scope.instance_events.extend(found)
"""Flow analysis.

Secret: how control flow inside one scope is approximated — which bindings
of a name can reach a statement (branches merge, ``return``/``raise``/
``break``/``continue`` end a path, loops run zero or one time, handlers see
any prefix of their ``try`` body) and which narrowing conditions
(``isinstance`` checks, ``is None`` tests, asserts) hold there.

A state maps a name to a set of facts ``(binding, constraints)``. The binding
``OUTER`` stands for "whatever the enclosing scope provides"; it only appears
when a name that is not bound locally gets narrowed.
"""

import ast

from . import scopes

OUTER = None


class Constraint:
    """A condition known to hold for a name: ``isinstance`` or ``is None``."""

    ISINSTANCE, NONE = "isinstance", "none"

    def __init__(self, kind, positive, expr=None):
        self.kind = kind
        self.positive = positive
        self.expr = expr

    def negated(self):
        return Constraint(self.kind, not self.positive, self.expr)


class State:
    def __init__(self, names=None, dead=False):
        self.names = names if names is not None else {}
        self.dead = dead

    def facts(self, name):
        return self.names.get(name)

    def bound(self, binding):
        names = dict(self.names)
        names[binding.name] = frozenset([(binding, ())])
        return State(names, self.dead)

    def narrowed(self, conditions):
        if not conditions:
            return self
        names = dict(self.names)
        for name, constraint in conditions:
            facts = names.get(name) or frozenset([(OUTER, ())])
            names[name] = frozenset((b, c + (constraint,)) for b, c in facts)
        return State(names, self.dead)

    def killed(self):
        return State(self.names, True)


def merge(states, keep_dead=False):
    """Join the states of paths meeting at one point."""
    states = [s for s in states if s is not None]
    live = [s for s in states if not s.dead]
    pool = states if keep_dead or not live else live
    if not pool:
        return State(dead=True)
    if len(pool) == 1:
        return State(pool[0].names, not live and not keep_dead)
    names = {}
    for state in pool:
        for name in state.names:
            names.setdefault(name, None)
    for name in names:
        facts = set()
        missing = False
        for state in pool:
            found = state.names.get(name)
            if found is None:
                missing = True
            else:
                facts.update(found)
        if missing and any(b is OUTER for b, _ in facts):
            facts.add((OUTER, ()))
        names[name] = frozenset(facts)
    return State(names, not live and not keep_dead)


class _Loop:
    def __init__(self):
        self.breaks = []
        self.continues = []


class Flow:
    """Reaching bindings at every statement of one scope."""

    def __init__(self, scope):
        self.scope = scope
        self.before = {}
        self.after = {}
        self._exits = []
        self._by_stmt = {}
        stray = []
        start = State()
        for event in scope.events:
            if not event.name or event.form == scopes.STAR:
                continue
            if event.form == scopes.PARAM:
                start = start.bound(event)
            elif event.home is scope and event.stmt is not None:
                self._by_stmt.setdefault(id(event.stmt), []).append(event)
            else:
                stray.append(event)
        body = scope.node.body if scope.node is not None else []
        last = self._block(body, start, None)
        end = merge([last] + self._exits, keep_dead=True)
        for event in stray:
            facts = set(end.names.get(event.name, ()))
            facts.add((event, ()))
            end.names[event.name] = frozenset(facts)
        self.end = end
        self.body = body

    # -- statements ------------------------------------------------------
    def _apply(self, node, state):
        for event in self._by_stmt.get(id(node), ()):
            state = state.bound(event)
        return state

    def _block(self, statements, state, loop):
        for statement in statements:
            self.before[id(statement)] = state
            state = self._statement(statement, state, loop)
            self.after[id(statement)] = state
        return state

    def _statement(self, node, state, loop):
        if isinstance(node, ast.If):
            state = self._apply(node, state)
            positive, negative = conditions(node.test)
            body = self._block(node.body, state.narrowed(positive), loop)
            other = self._block(node.orelse, state.narrowed(negative), loop)
            merged = merge([body, other])
            if not body.dead and not other.dead:
                merged = _forget(merged, positive, negative)
            return merged
        if isinstance(node, ast.While):
            state = self._apply(node, state)
            positive, negative = conditions(node.test)
            inner = _Loop()
            body = self._block(node.body, state.narrowed(positive), inner)
            if _always_true(node.test):
                normal = State(state.names, True)
            else:
                normal = merge([state, body] + inner.continues).narrowed(negative)
            other = self._block(node.orelse, normal, loop)
            return merge([other] + inner.breaks)
        if isinstance(node, (ast.For, ast.AsyncFor)):
            inner = _Loop()
            body = self._block(node.body, self._apply(node, state), inner)
            normal = merge([state, body] + inner.continues)
            other = self._block(node.orelse, normal, loop)
            return merge([other] + inner.breaks)
        if isinstance(node, (ast.With, ast.AsyncWith)):
            return self._block(node.body, self._apply(node, state), loop)
        if isinstance(node, (ast.Try, getattr(ast, "TryStar", ast.Try))):
            return self._try(node, state, loop)
        if isinstance(node, ast.Match):
            state = self._apply(node, state)
            ends = []
            exhaustive = False
            for case in node.cases:
                start = self._apply(case, state)
                self.before[id(case)] = start
                ends.append(self._block(case.body, start, loop))
                if case.guard is None and isinstance(case.pattern, ast.MatchAs) \
                        and case.pattern.pattern is None:
                    exhaustive = True
            if not exhaustive:
                ends.append(state)
            return merge(ends)
        if isinstance(node, (ast.Return, ast.Raise)):
            state = self._apply(node, state)
            self._exits.append(state)
            return state.killed()
        if isinstance(node, ast.Break):
            if loop is not None:
                loop.breaks.append(state)
            return state.killed()
        if isinstance(node, ast.Continue):
            if loop is not None:
                loop.continues.append(state)
            return state.killed()
        if isinstance(node, ast.Assert):
            state = self._apply(node, state)
            return state.narrowed(conditions(node.test)[0])
        return self._apply(node, state)

    def _try(self, node, state, loop):
        body = self._block(node.body, state, loop)
        partial = [state] + [self.after[id(s)] for s in node.body if id(s) in self.after]
        handler_start = merge(partial, keep_dead=True)
        ends = []
        for handler in node.handlers:
            start = self._apply(handler, handler_start)
            self.before[id(handler)] = start
            ends.append(self._block(handler.body, start, loop))
        ends.insert(0, self._block(node.orelse, body, loop))
        joined = merge(ends)
        if node.finalbody:
            start = merge(ends + [handler_start], keep_dead=True) if joined.dead else joined
            final = self._block(node.finalbody, start, loop)
            return State(final.names, joined.dead or final.dead)
        return joined

    # -- queries ---------------------------------------------------------
    def state_before(self, node):
        """State just before ``node`` (a statement, handler or case) runs."""
        return self.before.get(id(node))

    def state_at(self, line, col, indent):
        """State at a cursor that may sit between or after statements.

        ``indent`` is the indentation of the cursor's line (used to decide
        which block an empty or broken line belongs to).
        """
        return self._find(self.body, line, col, indent, self.end)

    def _find(self, statements, line, col, indent, fallback):
        previous = None
        for statement in statements:
            if (statement.lineno, statement.col_offset) > (line, col):
                break
            previous = statement
        if previous is None:
            return self.before.get(id(statements[0]), fallback) if statements else fallback
        blocks = _blocks(previous)
        if line <= (previous.end_lineno or previous.lineno):
            inner = [block for block in blocks
                     if (block[0].lineno, block[0].col_offset) <= (line, col)
                     or block[0].lineno < line]
            if inner:
                block = inner[-1]
                return self._find(block, line, col, indent,
                                  self.after.get(id(block[-1]), fallback))
            return self.before.get(id(previous), fallback)
        if blocks and indent > previous.col_offset:
            block = blocks[-1]
            return self._find(block, line, col, indent, self.after.get(id(block[-1]), fallback))
        return self.after.get(id(previous), fallback)


def _forget(state, positive, negative):
    """Drop narrowing by one test where both of its outcomes rejoin.

    A binding narrowed one way on the true path and the other way on the
    false path is, after the paths meet, just the binding again. (When one
    path rebinds the name, the other path's narrowing is kept.)
    """
    own = {id(c) for _, c in positive + negative}
    if not own:
        return state
    names = dict(state.names)
    for name, facts in state.names.items():
        groups = {}
        for binding, constraints in facts:
            base = tuple(c for c in constraints if id(c) not in own)
            groups.setdefault((binding, base), []).append(constraints)
        if all(len(variants) == 1 for variants in groups.values()):
            continue
        result = set()
        for (binding, base), variants in groups.items():
            if len(variants) > 1:
                result.add((binding, base))
            else:
                result.add((binding, variants[0]))
        names[name] = frozenset(result)
    return State(names, state.dead)


def _blocks(node):
    """Nested non-empty statement lists of a compound statement, in source order."""
    found = []
    for field in ("body", "handlers", "orelse", "finalbody", "cases"):
        value = getattr(node, field, None)
        if not value or not isinstance(value, list):
            continue
        if field in ("handlers", "cases"):
            found.extend(item.body for item in value if item.body)
        else:
            found.append(value)
    found.sort(key=lambda block: (block[0].lineno, block[0].col_offset))
    return found


def _always_true(test):
    return isinstance(test, ast.Constant) and bool(test.value)


def conditions(test):
    """``(holds_if_true, holds_if_false)``: lists of (name, Constraint)."""
    if isinstance(test, ast.UnaryOp) and isinstance(test.op, ast.Not):
        positive, negative = conditions(test.operand)
        return negative, positive
    if isinstance(test, ast.BoolOp):
        parts = [conditions(value) for value in test.values]
        if isinstance(test.op, ast.And):
            return [c for p, _ in parts for c in p], []
        return [], [c for _, n in parts for c in n]
    if isinstance(test, ast.NamedExpr):
        return conditions(test.value)
    if isinstance(test, ast.Call) and isinstance(test.func, ast.Name) \
            and test.func.id == "isinstance" and len(test.args) == 2 \
            and isinstance(test.args[0], ast.Name):
        constraint = Constraint(Constraint.ISINSTANCE, True, test.args[1])
        name = test.args[0].id
        return [(name, constraint)], [(name, constraint.negated())]
    if isinstance(test, ast.Compare) and len(test.ops) == 1:
        left, right = test.left, test.comparators[0]
        if _is_none(left):
            left, right = right, left
        if isinstance(left, ast.Name) and _is_none(right):
            op = test.ops[0]
            if isinstance(op, (ast.Is, ast.Eq)):
                positive = True
            elif isinstance(op, (ast.IsNot, ast.NotEq)):
                positive = False
            else:
                return [], []
            constraint = Constraint(Constraint.NONE, positive)
            return [(left.id, constraint)], [(left.id, constraint.negated())]
    return [], []


def _is_none(node):
    return isinstance(node, ast.Constant) and node.value is None
